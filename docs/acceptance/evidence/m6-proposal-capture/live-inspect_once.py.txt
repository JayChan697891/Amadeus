from dataclasses import asdict
from pathlib import Path
import ast
import hashlib
import json
import pilot

base=Path(__file__).resolve().parent
state=json.loads((base/"state.json").read_text(encoding="utf-8"))
core=pilot.Core(base/"core")
actor=pilot.actor_for(state,pilot.deny)
relay=pilot.ControlledRelay(core,base,base/"receipts")
outcome=relay.observe_codex_proposal(actor,state["task"]["id"])
auth={"project_id":state["owner"]["project_id"],"session_id":state["member"]["session_id"],"session_token":state["member"]["session_token"]}
project=core.execute("get_project_state",auth)
task=next(row for row in project["tasks"] if row["id"]==state["task"]["id"])
record={"runtime_sha":pilot.git(pilot.REPO,"rev-parse","HEAD"),"plan_sha256":state["plan_sha256"],"observation":asdict(outcome),"task_status":task["status"],"task_verification":task.get("task_verification","NOT_REPORTED"),"legacy_observation":asdict(relay.observe(actor,state["task"]["id"])),"model_calls":1,"retry":False,"code_executed":False,"code_imported":False,"artifact_verification":"NOT RUN","formal_release":"BLOCKED","legacy_title_note":"The inherited Core title is not the specification; the pinned launch instruction and JSON contract define this task."}
if outcome.state=="proposal_stored":
 context=core.execute("get_context",{**auth,"context_id":outcome.context_id,"version":outcome.context_version})
 content=context["content"]
 if hashlib.sha256(content.encode("utf-8")).hexdigest()!=outcome.content_sha256:raise RuntimeError("Context hash mismatch")
 envelope=json.loads(content)
 files=envelope["files"]
 if len(files)!=1 or files[0]["path"]!="add.py":raise RuntimeError("Proposal contract mismatch")
 source=files[0]["content"]
 record["proposal_file"]={"path":"add.py","utf8_bytes":len(source.encode("utf-8")),"sha256":hashlib.sha256(source.encode("utf-8")).hexdigest()}
 try:
  module=ast.parse(source)
  function=module.body[0] if len(module.body)==1 else None
  record["syntax_only_shape"]={"parsed":True,"one_add_function":isinstance(function,ast.FunctionDef) and function.name=="add","no_imports":not any(isinstance(node,(ast.Import,ast.ImportFrom)) for node in ast.walk(module)),"no_top_level_actions":all(isinstance(node,ast.FunctionDef) for node in module.body),"code_executed":False,"semantic_correctness":"UNVERIFIED"}
 except (SyntaxError,ValueError,RecursionError):
  record["syntax_only_shape"]={"parsed":False,"code_executed":False,"semantic_correctness":"UNVERIFIED"}
metadata=json.loads((base/"capture-metadata.json").read_text(encoding="utf-8"))
events=[];items=[]
if metadata["saved"]:
 raw=(base/"private-capture/stdout.bin").read_bytes()
 if len(raw)>1024*1024:raise RuntimeError("Capture bound exceeded")
 for line in raw.splitlines():
  try:event=json.loads(line)
  except (ValueError,UnicodeError):continue
  if isinstance(event,dict):
   event_type=event.get("type")
   events.append(event_type if event_type in {"thread.started","turn.started","item.started","item.updated","item.completed","turn.completed","turn.failed","error"} else "other")
   item=event.get("item")
   if isinstance(item,dict):
    item_type=item.get("type")
    items.append(item_type if item_type in {"agent_message","reasoning","command_execution","file_change","mcp_tool_call","web_search","todo_list","error"} else "other")
record["event_types"]=events;record["item_types"]=items
record["raw_uploaded"]=False
with (base/"reopened-check.json").open("x",encoding="utf-8") as stream:
 json.dump(record,stream,indent=2);stream.write("\n")
print(json.dumps(record,indent=2))
