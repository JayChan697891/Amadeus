"""Record only changed-file style and source typing checks; no tests."""
import ast
import hashlib
import json
import subprocess
import sys
from pathlib import Path

BASE = Path(__file__).parent
ROOT = Path("F:/syuc/amadeus-m6-single")
DEST = BASE / sys.argv[1]
DEST.mkdir(exist_ok=False)
FILES = ["src/agentsync/controlled_relay.py", "src/agentsync/relay_status_cli.py",
         "tests/integration/test_relay_owner_status.py"]
for name in FILES: ast.parse((ROOT/name).read_bytes(), filename=name)
commands = [[sys.executable,"-I","-B","-m","ruff","check","--no-cache",*FILES],
            [sys.executable,"-I","-B","-m","ruff","format","--check","--no-cache",*FILES],
            [sys.executable,"-I","-B","-m","mypy","--cache-dir",str(BASE/"mypy-cache"),"src/agentsync"]]
checks = []
for index, command in enumerate(commands):
    result = subprocess.run(command,cwd=ROOT,capture_output=True,check=False)
    with (DEST/f"{index}.stdout").open("xb") as stream: stream.write(result.stdout)
    with (DEST/f"{index}.stderr").open("xb") as stream: stream.write(result.stderr)
    checks.append({"argv":command,"exit_code":result.returncode})
with (DEST/"result.json").open("x",encoding="utf8") as stream:
    json.dump({"checks":checks,"source_sha256":{name:hashlib.sha256((ROOT/name).read_bytes()).hexdigest()
                                              for name in [*FILES,"docs/RELAY-RECOVERY.md"]}},stream,indent=2)
print(json.dumps({"exits":[check["exit_code"] for check in checks],"result":str(DEST/"result.json")}))
raise SystemExit(any(check["exit_code"] for check in checks))
