"""Install one reviewed a123 wheel offline and check seven affected files; no models."""
import hashlib
import json
import os
import re
import stat
import sys
import tomllib
import xml.etree.ElementTree as ET
import zipfile
from datetime import UTC, datetime
from pathlib import Path

BASE = Path("/root/amadeus-m123-live")
PYTHON = Path("/root/jobs-env/bin/python")
WHEELHOUSE = Path("/root/intent-wheelhouse")
COMMIT = "a792b6e759aa1990391f8195ba97676a94713f85"
RUNTIME = "bdd808b8e5cd1caa91ec94d69cb11de9aed4cee7"
VERSION = "0.3.0a123"
SOURCE, ENV, RUN, TEMP, HOME = (BASE / name for name in ("s", "w", "run", "t", "home"))
WHEEL = BASE / f"agentsync_local-{VERSION}-py3-none-any.whl"
SELECTION = [
    "tests/unit/test_codex_adapter.py", "tests/unit/test_process_capture.py",
    "tests/integration/test_codex_proposal_relay_queue.py", "tests/integration/test_unfinished_continuation.py",
    "tests/integration/test_verified_codex_terminal.py", "tests/unit/test_relay_execution_budget.py",
    "tests/unit/test_proposals.py",
]
ORIGIN = r'''
import hashlib,importlib.metadata as m,json,sys,sysconfig
from pathlib import Path
import agentsync,agentsync.cli
env,archive,members=map(Path,sys.argv[1:4]); expected=sys.argv[4]
site=Path(sysconfig.get_path('purelib')).resolve()
assert Path(sys.prefix).resolve()==env.resolve() and site.is_relative_to(env.resolve())
assert sys.version_info[:3]==(3,14,7)
assert Path(agentsync.__file__).resolve()==site/'agentsync/__init__.py'
assert Path(agentsync.cli.__file__).resolve()==site/'agentsync/cli.py'
dist=m.distribution('agentsync-local'); direct=json.loads(dist.read_text('direct_url.json'))
assert dist.version==agentsync.__version__=='0.3.0a123'
assert direct['url']==archive.resolve().as_uri() and 'dir_info' not in direct
assert direct['archive_info']['hashes']['sha256']==expected
actual={p.relative_to(site).as_posix():hashlib.sha256(p.read_bytes()).hexdigest()
        for p in (site/'agentsync').rglob('*') if p.is_file() and '__pycache__' not in p.parts}
assert actual==json.loads(members.read_bytes()) and len(actual)==97
print(json.dumps({'origin':str(Path(agentsync.__file__).resolve()),'version':dist.version,
                  'prefix':sys.prefix,'direct_url':direct,'runtime_members':len(actual)}))
'''

def sha(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()

def save(path, value):
    with path.open("x", encoding="utf8", newline="\n") as stream:
        json.dump(value, stream, indent=2)
        stream.write("\n")

def main():
    assert sys.platform == "linux" and sys.version_info[:3] == (3, 14, 7)
    assert Path(sys.executable).resolve() == PYTHON.resolve() and sys.flags.isolated and sys.dont_write_bytecode
    assert len(sys.argv) == 2 and re.fullmatch("[0-9a-f]{64}", sys.argv[1])
    assert sha(BASE / "bundle.json") == sys.argv[1]  # Root supplies the reviewed actual build digest.
    bundle = json.loads((BASE / "bundle.json").read_bytes())
    assert (bundle["source_commit"], bundle["runtime_commit"], bundle["version"], bundle["runtime_members"]) == (COMMIT, RUNTIME, VERSION, 97)
    expected_names = {"source.zip", "source-manifest.json", "runtime-members.json", WHEEL.name}
    assert set(bundle["inputs"]) == expected_names
    assert {p.name for p in BASE.iterdir()} <= expected_names | {"bundle.json", Path(__file__).name, "guest-preflight.json"}
    def verify_inputs():
        for name, expected in bundle["inputs"].items():
            path = BASE / name
            assert path.resolve(strict=True) == path and stat.S_ISREG(path.lstat().st_mode)
            assert (path.stat().st_size, sha(path)) == (expected["size"], expected["sha256"])
    verify_inputs()
    manifest = json.loads((BASE / "source-manifest.json").read_bytes())
    members = json.loads((BASE / "runtime-members.json").read_bytes())
    assert len(manifest) == bundle["source_files"] and len(members) == 97
    assert all(row.get("git_mode") in ("100644", "100755") for row in manifest.values())
    assert members == {name.removeprefix("src/"): row["sha256"] for name, row in manifest.items() if name.startswith("src/agentsync/")}
    for path in (SOURCE, RUN, TEMP, HOME): path.mkdir(exist_ok=False)
    (RUN / "logs").mkdir()
    for key in tuple(os.environ):
        if key.upper().startswith(("PYTHON", "PYTEST_", "PIP_", "UV_", "AGENTSYNC_", "OPENAI_",
                                   "ANTHROPIC_", "CODEX_", "CLAUDE_", "CURSOR_", "GIT_", "GITHUB_")):
            os.environ.pop(key)
    os.environ.update(HOME=str(HOME), USERPROFILE=str(HOME), XDG_CONFIG_HOME=str(HOME),
                      XDG_CACHE_HOME=str(HOME), XDG_DATA_HOME=str(HOME), TEMP=str(TEMP), TMP=str(TEMP), TMPDIR=str(TEMP),
                      PYTHONDONTWRITEBYTECODE="1", PYTHONNOUSERSITE="1", PIP_CONFIG_FILE=os.devnull,
                      PIP_NO_INDEX="1", PIP_NO_CACHE_DIR="1", PIP_DISABLE_PIP_VERSION_CHECK="1",
                      GIT_CONFIG_NOSYSTEM="1", GIT_CONFIG_GLOBAL=os.devnull, GIT_TERMINAL_PROMPT="0")
    result = {"status": "fail", "version": VERSION, "source_commit": COMMIT, "runtime_commit": RUNTIME,
              "bundle_sha256": sys.argv[1], "controller_pid": os.getpid(), "driver_sha256": sha(Path(__file__)),
              "started_at": datetime.now(UTC).isoformat(), "boot_id": Path("/proc/sys/kernel/random/boot_id").read_text().strip(),
              "scope": "Linux3.14 installed wheel, seven affected files; deterministic fixtures only",
              "model_calls": 0, "real_clients": "NOT RUN", "candidate_execution": "NOT RUN",
              "docker": "NOT RUN; opt-in environment cleared, existing fixture skips retained",
              "full_suite": "NOT RUN", "formal_release": "BLOCKED", "automatic_retry": False, "checks": []}
    save(RUN / "start.json", result)
    collector = None
    raw = TEMP / "focused-raw.xml"
    def verify_source():
        actual = {p.relative_to(SOURCE).as_posix(): {"size": p.stat().st_size, "sha256": sha(p)}
                  for p in SOURCE.rglob("*") if p.is_file()}
        assert actual == {name: {"size": row["size"], "sha256": row["sha256"]} for name, row in manifest.items()}
    try:
        with zipfile.ZipFile(BASE / "source.zip") as tree:
            assert tree.comment == COMMIT.encode()
            files = [item for item in tree.infolist() if not item.is_dir()]
            assert len(files) == len(manifest) and {item.filename for item in files} == set(manifest)
            for item in files:
                name = item.filename
                assert not name.startswith("/") and ".." not in Path(name).parts and ":" not in name and "\\" not in name
                mode = item.external_attr >> 16
                assert stat.S_ISREG(mode) or (item.create_system == 0 and item.external_attr == 0 and manifest[name]["git_mode"] in ("100644", "100755"))
                target = SOURCE / name
                target.parent.mkdir(parents=True, exist_ok=True)
                with tree.open(item) as incoming, target.open("xb") as outgoing:
                    while data := incoming.read(1024 * 1024): outgoing.write(data)
        verify_source()
        sys.path.insert(0, str(SOURCE))
        from tools import release_acceptance as collector
        project = tomllib.loads((SOURCE / "pyproject.toml").read_text("utf8"))
        packages = collector._prepare_offline_packages(project, WHEELHOUSE, TEMP, RUN)
        def execute(name, argv, capture=None):
            check = collector.execute(name, list(map(str, argv)), RUN, RUN, TEMP,
                                      offline_wheelhouse=packages.wheelhouse, capture_stdout=capture)
            result["checks"].append(check)
            assert check["status"] == "pass" and check["exit_code"] == 0 and not check["log_truncated"]
        python = ENV / "bin/python"
        origin = [python, "-I", "-B", "-c", ORIGIN, ENV, WHEEL, BASE / "runtime-members.json", bundle["inputs"][WHEEL.name]["sha256"]]
        execute("wheel_venv", [PYTHON, "-I", "-B", "-m", "venv", ENV])
        execute("wheel_offline_dependencies", collector._offline_install(str(python), packages.wheelhouse, "--only-binary=:all:", *packages.application))
        execute("wheel_install", collector._offline_install(str(python), packages.wheelhouse, "--no-deps", "--", str(WHEEL)))
        execute("wheel_pip_check", [python, "-I", "-B", "-m", "pip", "check"])
        execute("wheel_origin_before", origin)
        try:
            execute("wheel_pytest", [python, "-I", "-B", "-m", "pytest", "-q", "-p", "no:cacheprovider", "--tb=short",
                                    "--basetemp", TEMP / "pytest", "--junitxml", raw, *(SOURCE / name for name in SELECTION)], RUN / "stdout.private")
            result["status"] = "pass"
        finally:
            try:
                execute("wheel_origin_after", origin)
            except Exception as error:
                result.update(status="fail", origin_after_error_type=type(error).__name__)
            if raw.is_file():
                try:
                    redactions = collector.persist_junit(raw, RUN / "focused-junit.xml")
                    cases = {}
                    for case in ET.parse(RUN / "focused-junit.xml").iter("testcase"):
                        key = (case.get("classname"), case.get("name"))
                        assert key not in cases
                        cases[key] = next((kind for kind in ("failure", "error", "skipped") if case.find(kind) is not None), "pass")
                    result["junit"] = {"sha256": sha(RUN / "focused-junit.xml"), "raw_private_sha256": sha(raw),
                                       "credential_redactions": redactions, "unique_cases": len(cases),
                                       "outcomes": {kind: sum(value == kind for value in cases.values()) for kind in ("pass", "failure", "error", "skipped")}}
                except Exception as error:
                    result.update(status="fail", junit_error_type=type(error).__name__)
            else:
                result.update(status="fail", junit_error_type="MissingJUnit")
    except Exception as error:
        result.update(status="fail", error_type=type(error).__name__)
        raise
    finally:
        try:
            verify_inputs()
            verify_source()
        except Exception as error:
            result.update(status="fail", postcheck_error_type=type(error).__name__)
        stdout = RUN / "stdout.private"
        if stdout.is_file() and collector is not None:
            try:
                with stdout.open("rb") as stream: data = stream.read(collector.MAX_LOG_BYTES + 1)
                clean, redactions = collector.redact_credentials(data[:collector.MAX_LOG_BYTES].decode("utf8", errors="replace"))
                with (RUN / "stdout.log").open("x", encoding="utf8") as stream: stream.write(clean)
                result["stdout"] = {"sha256": sha(RUN / "stdout.log"), "raw_private_sha256": sha(stdout), "credential_redactions": redactions}
                if len(data) > collector.MAX_LOG_BYTES: result.update(status="fail", stdout_error_type="OutputTruncated")
            except Exception as error:
                result.update(status="fail", stdout_error_type=type(error).__name__)
        save(RUN / "result.json", result)
    print(json.dumps({"status": result["status"], "junit": result.get("junit"), "model_calls": 0}))
    return 0 if result["status"] == "pass" else 1

if __name__ == "__main__":
    raise SystemExit(main())
