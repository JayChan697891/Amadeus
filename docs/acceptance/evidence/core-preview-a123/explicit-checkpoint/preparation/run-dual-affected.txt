"""One installed-development regression attempt; prepared only until Root launches."""
import json
import os
import runpy
import sys
from datetime import datetime, timezone
from pathlib import Path

BASE = Path(__file__).parent
helpers = runpy.run_path(str(BASE/'prepare_install_r3.py'))
SOURCE, PYTHON = helpers['SOURCE'], helpers['PYTHON']
sha, save, verify = (helpers[name] for name in ('sha','save','verify'))
SELECTION = [
    'tests/unit/test_codex_adapter.py',
    'tests/unit/test_process_capture.py',
    'tests/integration/test_codex_proposal_relay_queue.py',
    'tests/integration/test_unfinished_continuation.py',
    'tests/integration/test_verified_codex_terminal.py',
    'tests/unit/test_relay_execution_budget.py',
    'tests/unit/test_proposals.py',
]

def main():
    assert Path(sys.executable).resolve()==PYTHON.resolve()
    assert sys.version_info[:3]==(3,12,16) and sys.flags.isolated and sys.dont_write_bytecode
    install=json.loads((BASE/'install-r3/result.json').read_bytes())
    assert install['status']=='pass' and install['tests']=='NOT RUN'
    manifest=json.loads((BASE/'source-manifest-r3.json').read_bytes())
    assert sha(BASE/'source-manifest-r3.json')==install['source_manifest_sha256']
    assert sha(BASE/'runtime-members-r3.json')==install['runtime_members_sha256']
    verify(manifest)
    run,temp,home=(BASE/name for name in ('dual-affected','dual-t','dual-home'))
    for directory in (run,temp,home): directory.mkdir(exist_ok=False)
    (run/'logs').mkdir()
    helpers['isolate'](home,temp)
    tool=helpers['collector']()
    raw=temp/'affected-raw.xml'
    command=[str(PYTHON),'-I','-B','-m','pytest','-p','no:cacheprovider','--tb=short',
             '--basetemp',str(temp/'pytest'),'--junitxml',str(raw),*SELECTION]
    start={'version':'0.3.0a123','controller_pid':os.getpid(),
           'started_at':datetime.now(timezone.utc).isoformat(),'argv':command,
           'driver_sha256':sha(Path(__file__)),'installation_receipt_sha256':sha(BASE/'install-r3/result.json'),
           'source_manifest_sha256':sha(BASE/'source-manifest-r3.json'),
           'runtime_members_sha256':sha(BASE/'runtime-members-r3.json'),
           'scope':'Seven affected files; installed development runtime; deterministic owned Python/Core fixtures only',
           'model_calls':0,'real_clients':'NOT RUN','docker':'NOT RUN; platform skips retained',
           'full_suite':'NOT RUN','formal_release':'BLOCKED','automatic_retry':False}
    save(run/'start.json',start)
    result={**start,'status':'fail','checks':[]}
    origin=[str(PYTHON),'-I','-B','-c',helpers['PROBE'],str(BASE/'venv'),str(SOURCE),str(BASE/'runtime-members-r3.json')]
    def execute(name,argv,capture=None):
        check=tool.execute(name,argv,SOURCE,run,temp,capture_stdout=capture)
        result['checks'].append(check)
        assert check['status']=='pass' and check['exit_code']==0 and not check['log_truncated']
    try:
        execute('origin_before',origin)
        # This ID deliberately is not 'pytest': the collector must not set source PYTHONPATH.
        execute('affected_pytest',command,run/'stdout.private')
        result['status']='pass'
    except Exception as error:
        result['error_type']=type(error).__name__
        raise
    finally:
        try:
            execute('origin_after',origin)
            verify(manifest)
        except Exception as error:
            result['status']='fail'
            result['postcheck_error_type']=type(error).__name__
        stdout=run/'stdout.private'
        if stdout.is_file():
            try:
                with stdout.open('rb') as stream: data=stream.read(tool.MAX_LOG_BYTES+1)
                clean,redactions=tool.redact_credentials(data[:tool.MAX_LOG_BYTES].decode('utf8',errors='replace'))
                with (run/'stdout.log').open('x',encoding='utf8',newline='\n') as stream: stream.write(clean)
                result['stdout']={'raw_private_sha256':sha(stdout),'public_sha256':sha(run/'stdout.log'),
                                  'credential_redactions':redactions,'truncated':len(data)>tool.MAX_LOG_BYTES}
                if result['stdout']['truncated']: result['status']='fail'
            except Exception as error:
                result['status']='fail'
                result['stdout_error_type']=type(error).__name__
        if raw.is_file():
            try:
                redactions=tool.persist_junit(raw,run/'affected-junit.xml')
                result['junit']={'raw_private_path':str(raw),'raw_sha256':sha(raw),
                                 'public_sha256':sha(run/'affected-junit.xml'),'credential_redactions':redactions}
            except Exception as error:
                result['status']='fail'
                result['junit_error_type']=type(error).__name__
        else:
            result['status']='fail'
            result['junit_error_type']='MissingJUnit'
        save(run/'result.json',result)
    print(json.dumps({'status':result['status'],'result':str(run/'result.json')}))
    return 0 if result['status']=='pass' else 1

if __name__=='__main__':
    raise SystemExit(main())
