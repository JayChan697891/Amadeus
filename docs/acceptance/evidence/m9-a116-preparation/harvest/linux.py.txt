"""Copy one completed local VM row, preserving original bytes and strict gates."""
import argparse
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path, PurePosixPath
import stat
import sys

import paramiko

parser = argparse.ArgumentParser()
parser.add_argument("--row", choices=("linux-314", "linux-311", "linux-312"), required=True)
args = parser.parse_args()
base = Path("D:/agentsync-debug/m116")
destination = base / "harvest" / args.row
assert not destination.exists()
verifier_sha = "6abb5a549566a6c9c90279e9474814336fdc0a166b5641d609d0f92e4883187c"
assert hashlib.sha256((base / "s/tools/acceptance_evidence.py").read_bytes()).hexdigest() == verifier_sha
sys.path.insert(0, str(base / "s"))
from tools.acceptance_evidence import _member_parts, read_json, verify_report_evidence

remote_row = "/root/amadeus-m116/" + args.row
client = paramiko.SSHClient()
client.load_host_keys("D:/agentsync-debug/vm/known_hosts")
client.set_missing_host_key_policy(paramiko.RejectPolicy())
client.connect("127.0.0.1", port=22222, username="root",
               key_filename="D:/agentsync-debug/vm/debug_ed25519",
               allow_agent=False, look_for_keys=False, timeout=15)
copied = []
total = 0
try:
    sftp = client.open_sftp()
    sftp.get_channel().settimeout(30)

    def read(remote):
        before = sftp.lstat(remote)
        assert stat.S_ISREG(before.st_mode) and before.st_size <= 32 * 1024 * 1024
        with sftp.open(remote, "rb") as stream:
            data = stream.read(32 * 1024 * 1024 + 1)
        after = sftp.lstat(remote)
        assert len(data) == before.st_size == after.st_size
        assert before.st_mtime == after.st_mtime
        return data

    result_bytes = read(remote_row + "/result.json")
    result = read_json(result_bytes)
    assert result["phase"] == "finished" and result["local_checks_passed"] is True
    assert result["strict_evidence_verified"] is True and result["source_clean"] is True
    assert result["docker_containers_after"] == []
    assert result["commit_sha"] == "624b36c039a90feb9340559ac773f9df08ba86ad"
    assert result["candidate_manifest_sha256"] == "23365b89c3c4c66acabf2c51d5f9d7d90de02a1dec7664598a9ebc32c0cf9b95"
    assert result["driver_sha256"] == "765f252f1855054c3e6d0b6458c421902b4c8c7eb598b6e19208709e78a5e27d"
    remote_report = PurePosixPath(result["report"])
    assert remote_report.is_absolute() and ".." not in remote_report.parts
    assert remote_report.is_relative_to(PurePosixPath(remote_row)/"reports")
    assert remote_report.name == "report.json"
    remote_report_root = str(remote_report.parent)
    terminal_bytes = read("/root/amadeus-m116/linux-queue/"+args.row+"-terminal.json")
    terminal = read_json(terminal_bytes)
    assert terminal["row"] == args.row and terminal["exit_code"] == 0
    assert terminal["local_checks_passed"] is True and terminal["strict_evidence_verified"] is True
    report_bytes = read(result["report"])
    report = read_json(report_bytes)
    assert all(check["status"] == "pass" and check["exit_code"] == 0 for check in report["checks"])
    destination.mkdir(parents=True)

    def save(name, data):
        global total
        parts = _member_parts(name)
        assert "private-junit" not in parts
        total += len(data)
        assert total <= 100 * 1024 * 1024
        path = destination.joinpath(*parts)
        path.parent.mkdir(parents=True, exist_ok=True)
        with path.open("xb") as stream:
            stream.write(data)
        copied.append({"path": name, "size": len(data), "sha256": hashlib.sha256(data).hexdigest()})

    save("report.json", report_bytes)
    for entry in report["artifacts"]:
        name = entry["path"]
        _member_parts(name)
        assert "private-junit" not in name.split("/")
        data = read(remote_report_root + "/" + name)
        assert len(data) == entry["size"] and hashlib.sha256(data).hexdigest() == entry["sha256"]
        save(name, data)
    save("execution/result.json", result_bytes)
    save("execution/attempt.json", read(remote_row + "/attempt.json"))
    save("execution/queue-terminal.json", terminal_bytes)
    # The frozen tests emit these explicit, separate Docker observation groups.
    groups = {"verifier": 14, "owner_executor": 7, "reconciliation": 5,
              "candidate_executor": 5, "task_completion": 1,
              "verified_review": 1, "proposal_pipeline": 2}
    forbidden_keys = {"token", "access_token", "refresh_token", "api_key", "password",
                      "authorization", "cookie", "secret", "session_token", "bearer_token", "auth"}

    def check_keys(value):
        if isinstance(value, dict):
            for key, item in value.items():
                assert key.casefold() not in forbidden_keys or item in (None, "", False)
                check_keys(item)
        elif isinstance(value, list):
            for item in value:
                check_keys(item)

    def copy_observations(relative, depth=0):
        assert depth <= 2
        remote = remote_row + "/observations/" + relative
        assert stat.S_ISDIR(sftp.lstat(remote).st_mode)
        count = 0
        for entry in sorted(sftp.listdir_attr(remote), key=lambda item: item.filename):
            name = relative + "/" + entry.filename
            _member_parts(name)
            if stat.S_ISDIR(entry.st_mode):
                count += copy_observations(name, depth + 1)
            else:
                assert stat.S_ISREG(entry.st_mode) and entry.filename.endswith(".json")
                data = read(remote_row + "/observations/" + name)
                check_keys(read_json(data))
                save("observations/" + name, data)
                count += 1
        return count

    observation_counts = {}
    for suite in ("pytest", "wheel_pytest", "sdist_pytest"):
        for group, expected in groups.items():
            relative = suite + "/agentsync_" + group + "_evidence_dir"
            observed = copy_observations(relative)
            assert observed == expected, (relative, observed, expected)
            observation_counts[relative] = observed
    sftp.close()
finally:
    client.close()

audit = verify_report_evidence(destination / "report.json", report, source_repo=base / "s")
assert audit["junit_case_sets_match"] is True
save("transfer-audit.json", (json.dumps(audit, indent=2) + "\n").encode())
summary = {"at": datetime.now(timezone.utc).isoformat(), "remote": remote_report_root,
           "row": args.row, "files": copied, "copied_bytes": total,
           "docker_observation_counts": observation_counts,
           "docker_scope": "Actual isolated Docker with known local fixtures; expected negative outcomes retained; no real models",
           "scope": "Exact completed row transfer and strict evidence verification; not formal release",
           "raw_xml_uploaded": False, "formal_release": "BLOCKED"}
with (destination / "transfer-summary.json").open("x", encoding="utf-8") as stream:
    json.dump(summary, stream, indent=2)
print(json.dumps({"row": args.row, "files_verified": len(copied), "bytes": total,
                  "junit_case_sets_match": True, "formal_release": "BLOCKED"}))
