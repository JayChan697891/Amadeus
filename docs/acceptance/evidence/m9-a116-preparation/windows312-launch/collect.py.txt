def git(*args):
    return subprocess.check_output(['git', '-C', str(source), *args], timeout=30).decode().strip()

def write(path, value):
    with path.open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')

assert git('rev-parse', 'HEAD') == commit
assert git('rev-parse', 'HEAD^{tree}') == tree

sys.path.insert(0, str(source))

snapshot = collector.source_snapshot(source)
audit = verify_git_source(source, commit, tree, snapshot)
state = {'phase': 'started', 'pid': os.getpid(), 'started_at': datetime.now(UTC).isoformat(),
