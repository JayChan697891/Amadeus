def timestamp():
    return datetime.now(UTC).isoformat()


def sha(path):
    with path.open('rb') as stream:
        return hashlib.file_digest(stream, 'sha256').hexdigest()


def text_sha(text):
    return hashlib.sha256(text.encode('utf-8')).hexdigest()


def save(name, data):
    with (BASE / name).open('x', encoding='utf-8') as out:
        json.dump(data, out, indent=2)
        out.write('\n')


def load(name):
    return json.loads((BASE / name).read_text(encoding='utf-8'))


def private_directory(path):
    info = path.lstat()
    assert stat.S_ISDIR(info.st_mode) and info.st_uid == 0 and stat.S_IMODE(info.st_mode) == 0o700


def runtime():
    config = json.loads((PACKAGE / 'config.json').read_text(encoding='utf-8'))
    assert config['product_sha'] == RECEIVER_PRODUCT and config['wheel_sha256'] == WHEEL_SHA
    assert sha(PACKAGE / 'agentsync_local-0.3.0a109-py3-none-any.whl') == WHEEL_SHA
    site = Path(agentsync.__file__).resolve().parent.parent
    assert site.is_relative_to(PACKAGE / 'venv')
    members = config['runtime_members']
    assert len(members) == 89 and sum(name.endswith('.py') for name in members) == 88
    for name, expected in members.items():
        path = (site / name).resolve(strict=True)
        assert path.is_relative_to(site / 'agentsync') and sha(path) == expected
    for name, module in tuple(sys.modules.items()):
        if name == 'agentsync' or name.startswith('agentsync.'):
            assert Path(module.__file__).resolve().is_relative_to(site / 'agentsync')
    return {'runtime_package_members_matched': len(members), 'runtime_python_modules_matched': 88}


def core_and_relay():
    core = Core(PRODUCER_BASE / 'core')
    return core, ControlledRelay(core, PRODUCER_BASE, PRODUCER_BASE / 'receipts')


def auth(state, who='worker'):
    member = state[who]
    return {'project_id': state['owner']['project_id'], 'session_id': member['session_id'],
            'session_token': member['session_token']}


def current_task(core, state, task_id, who='worker'):
    tasks = core.execute('list_tasks', auth(state, who))['tasks']
    return next(task for task in tasks if task['id'] == task_id)


def task_summary(task):
    return {key: task[key] for key in ('id', 'status', 'version', 'task_verification')}


def candidate_counts(core):
    with connect(core.db_path) as db:
        return {name: db.execute('SELECT COUNT(*) FROM ' + name).fetchone()[0] for name in
                ('merge_candidates', 'candidate_verification_jobs', 'task_verifications')}


def git(root, *args):
    return subprocess.run(['/usr/bin/git', '-C', str(root), *args], check=True, capture_output=True,
                          text=True, timeout=10).stdout.strip()


def intact(state):
    roots = [PRODUCER_BASE / 'source', Path(state['producer_tree'])]
    if 'tree' in state:
        roots.append(Path(state['tree']['path']))
    for root in roots:
        if (sha(root / 'fixture.txt') != state['fixture_sha256']
                or git(root, 'rev-parse', 'HEAD') != state['baseline_head']
                or git(root, 'status', '--porcelain', '--untracked-files=all')):
            return False
    return True


def producer_observation(core, state):
    task = current_task(core, state, PRODUCER_TASK)
    assert task_summary(task) == {'id': PRODUCER_TASK, 'status': 'completed', 'version': 6,
                                  'task_verification': 'verified'}
    context = core.execute('get_context', {**auth(state), 'context_id': PRODUCER_CONTEXT, 'version': 1})
    assert (context['id'], context['project_id'], context['version'], context['kind'], context['source']) == (
        PRODUCER_CONTEXT, state['owner']['project_id'], 1, 'note', 'amadeus:task-proposal')
    assert context['author_agent_id'] == state['producer_agent_id']
    assert context['author_agent_id'] not in (state['worker']['agent_id'], state['reviewer']['agent_id'])
    assert text_sha(context['content']) == PRODUCER_CONTEXT_SHA
    value = json.loads(context['content'])
    assert value['task_id'] == PRODUCER_TASK and value['execution_provenance'] == 'unverified'
    files = parse_proposal(json.dumps({'files': value['files']}, ensure_ascii=False), ['solution.py'])
    text = files[0]['content']
    assert len(text.encode('utf-8')) == 153 and text_sha(text) == SOURCE_SHA
    return task, context


async def deny(_review):
    return None


def actor_for(state, authority=deny):
    worker, tree = state['worker'], state['tree']
    profile = CodexHostProfile(state['owner']['project_id'], worker['session_id'], EXE,
        Path(tree['path']), BINARY_SHA, expected_cli_version=CLI_VERSION, max_timeout_seconds=TIMEOUT,
        output_limit=OUTPUT_LIMIT, linux_readonly=True)
    env = {'PATH': '/usr/bin:/bin', 'HOME': str(HOME), 'CODEX_HOME': str(HOME),
           'TEMP': str(HOME), 'TMP': str(HOME)}
    return RelayActor(worker['agent_id'], worker['session_token'], profile, tree['worktree_id'],
                      tree['head_sha'], env, authority)


def item_for(state, actor):
    pin = ProposalContextPin(PRODUCER_TASK, PRODUCER_CONTEXT, 1, PRODUCER_CONTEXT_SHA)
    return CodexProposalQueueItem(actor, CodexProposalInput(state['task']['id'], state['task']['version'],
        INSTRUCTION, ('review.md',), TIMEOUT, submit_for_review=True, input_context=pin))


def exact_preview(runner, state, actor, source):
    item = item_for(state, actor)
    queue = RelayQueue(runner)
    token = state['owner']['admin_token']
    selection = queue.preview((item,), admin_token=token, parallel_limit=1)
    assert selection.state == 'selected' and len(selection.tasks) == 1 and selection.selection_sha256
    assert selection.tasks[0].task_id == state['task']['id']
    wave = queue._key(actor, wave_request(state))
    task = runner._single_codex_task(actor, item.task, token, wave_key=wave)
    plan = prepare_launch_review(actor.profile, task, environment=actor.environment)
    contract, body = task.instruction.split('Host contract: ', 1)[1].split('\n\n', 1)
    assert json.loads(contract)['input_context'] == asdict(item.task.input_context)
    assert 'Untrusted producer proposal' in body.split('\n', 1)[0]
    assert json.loads(body.split('\n', 1)[1]) == source['content']
    assert plan.plan.timeout_seconds == TIMEOUT and plan.plan.output_limit == OUTPUT_LIMIT
    assert 'OPENAI_API_KEY' not in plan.plan.child_env()
    return selection.selection_sha256, wave, plan


def helpers():
    return {name: sha(BASE / name) for name in ('pilot.py', 'private_capture.py', 'credential_home.py', 'observe.py')}


def pinned_state(core):
    state = load('state.json')
    assert helpers() == state['helper_sha256'] and sha(EXE) == BINARY_SHA and intact(state)
    source_task, source = producer_observation(core, state)
    assert digest(source_task) == state['producer_task_sha256']
    assert candidate_counts(core) == state['candidate_counts']
    return state, source


def register(core, owner, name, role):
    member = core.execute('register_agent', {**owner, 'name': name})
    members = core.execute_owner('list_members', {**owner, 'limit': 100, 'offset': 0})['members']
    current = next(entry for entry in members if entry['agent_id'] == member['agent_id'])
    core.execute_owner('set_agent_roles', {**owner, 'agent_id': member['agent_id'], 'roles': [role],
        'expected_version': current['version'], 'request_id': 'model-review-role-' + member['agent_id']})
    return member



def handoff_request(state):
    worker = state['worker']
    receiver = HandoffReceiver(state['owner']['project_id'], worker['agent_id'],
                               worker['session_id'], worker['session_token'])
    return ProposalHandoffRequest(PRODUCER_TASK, receiver, REQUEST, INSTRUCTION,
                                  ('review.md',), TIMEOUT, 'Review the existing JSON-lines sum proposal')


def wave_request(state):
    key = digest(['proposal-handoff-v1', state['owner']['project_id'], REQUEST])
    return 'handoff-' + key + '-wave'


def coordinator(core):
    return ProposalHandoff(core, PRODUCER_BASE, PRODUCER_BASE / 'receipts')


class PreparationReady(Exception):
    pass


async def prepare_coordinator(core, runner, state, source):
    def prepare_receiver(receiver, workspace):
        assert receiver == handoff_request(state).receiver
        state['task'] = current_task(core, state, workspace.task_id)
        state['tree'] = core.execute('inspect_owned_worktree', {**auth(state),
            'task_id': workspace.task_id, 'worktree_id': workspace.worktree_id})
        assert state['tree']['path'] == str(workspace.path)
        assert state['task']['version'] == workspace.task_version
        core.execute('heartbeat', auth(state))
        selection, wave, plan = exact_preview(runner, state, actor_for(state), source)
        state.update(selection_sha256=selection, wave_id=wave, plan_sha256=plan.plan_sha256)
        save('private-plan.json', {'argv': plan.plan.argv, 'stdin_text': plan.plan.stdin_text,
            'environment': plan.plan.child_env(), 'cwd': str(plan.plan.cwd)})
        # A deliberate checkpoint before the coordinator writes any dispatch intent.
        raise PreparationReady
    try:
        await coordinator(core).run(handoff_request(state), admin_token=state['owner']['admin_token'],
            build_receiver=prepare_receiver, source_outcome=SOURCE_OUTCOME, wait_seconds=0, budget=30)
    except PreparationReady:
        pass
    else:
        raise AssertionError('Expected the exact-plan review checkpoint')
    prefix = wave_request(state).removesuffix('-wave')
    assert not (PRODUCER_BASE / 'receipts' / (prefix + '-dispatch.json')).exists()
    attempts = runner.owner_inspect_attempts(state['owner']['project_id'], state['task']['id'],
                                             admin_token=state['owner']['admin_token'])
    assert all(item.state == 'not_dispatched' for item in attempts)
    assert RelayQueue(runner).observe(actor_for(state), wave_request(state)).state == 'not_started'
    save('coordinator-preparation.json', {'timestamp': timestamp(),
        'source_result_sha256': PRODUCER_RESULT_SHA, 'source_reference_derived_from_saved_result': True,
        'task_created_by_coordinator': True, 'worktree_created_by_coordinator': True,
        'dispatch_intent_absent': True, 'attempts_not_dispatched': True,
        'checkpoint': 'exact_plan_review_before_dispatch', 'model_calls': 0})


def prepare():
    members = runtime()
    assert sha(EXE) == BINARY_SHA
    private_directory(PRODUCER_BASE)
    legacy_path = PRODUCER_BASE / 'state.json'
    legacy_info = legacy_path.lstat()
    assert stat.S_ISREG(legacy_info.st_mode) and legacy_info.st_uid == 0
    assert stat.S_IMODE(legacy_info.st_mode) == 0o600 and 0 < legacy_info.st_size <= 65536
    old = json.loads(legacy_path.read_text(encoding='utf-8'))
    assert old['task']['id'] == PRODUCER_TASK and old['binary_sha256'] == BINARY_SHA
    save('prepare-entry.json', {'timestamp': timestamp(), 'model_calls': 0})
    HOME.mkdir(mode=0o700)
    core, runner = core_and_relay()
    owner = old['owner']
    state = {'owner': owner, 'producer_agent_id': old['producer']['agent_id'],
             'producer_tree': old['tree']['path'], 'fixture_sha256': old['fixture_sha256'],
             'baseline_head': old['baseline_head'], 'helper_sha256': helpers()}
    state['worker'] = register(core, owner, 'Codex pinned proposal review worker', 'worker')
    state['reviewer'] = register(core, owner, 'Independent host review observer', 'reviewer')
    source_task, source = producer_observation(core, state)
    assert state['worker']['agent_id'] != state['reviewer']['agent_id']
    state['producer_task_sha256'] = digest(source_task)
    state['candidate_counts'] = candidate_counts(core)
    asyncio.run(prepare_coordinator(core, runner, state, source))
    assert intact(state)
    selection, wave, plan = exact_preview(runner, state, actor_for(state), source)
    assert (selection, wave, plan.plan_sha256) == (state['selection_sha256'], state['wave_id'], state['plan_sha256'])
    save('state.json', state)
    save('plan.json', {'producer_product_sha': PRODUCER_PRODUCT, 'receiver_product_sha': RECEIVER_PRODUCT,
        'wheel_sha256': WHEEL_SHA, 'binary_sha256': BINARY_SHA, 'client_version': CLI_VERSION,
        'producer_task': task_summary(source_task), 'producer_context_id': PRODUCER_CONTEXT,
        'producer_context_version': 1, 'producer_context_sha256': PRODUCER_CONTEXT_SHA,
        'source_sha256': SOURCE_SHA, 'source_utf8_bytes': 153, 'review_task_id': state['task']['id'],
        'review_task_initial_version': state['task']['version'], 'plan_sha256': plan.plan_sha256,
        'selection_sha256': selection, 'wave_id': wave, 'queue_request': wave_request(state), 'allowed_paths': ['review.md'],
        'timeout_seconds': TIMEOUT, 'output_limit': OUTPUT_LIMIT, 'execution_budget_seconds': EXECUTION_BUDGET,
        'helper_sha256': state['helper_sha256'], 'source_and_worktrees_unchanged': True,
        'model_calls': 0, 'credential_copy': False, **members})
    print(json.dumps({'prepared': True, 'plan_sha256': plan.plan_sha256, 'model_calls': 0}))


def current_pins(runner, state, actor, source):
    selection, wave, plan = exact_preview(runner, state, actor, source)
    assert selection == state['selection_sha256'] and wave == state['wave_id']
    assert plan.plan_sha256 == state['plan_sha256']
    return plan


def ready_once(expected):
    runtime()
    core, runner = core_and_relay()
    state, source = pinned_state(core)
    assert expected == state['plan_sha256']
    assert not any((BASE / name).exists() for name in ('attempt.json', 'execute-entry.json'))
    private_directory(HOME)
    require_mount()
    assert not any(HOME.iterdir())
    save('readiness-entry.json', {'timestamp': timestamp(), 'model_calls': 0, 'retry_allowed': False})
    core.execute('heartbeat', auth(state))
    actor = actor_for(state)
    snapshot = runner._project_state(actor)
    session = next(row for row in snapshot['sessions'] if row['id'] == state['worker']['session_id'])
    assert session['online'] is True
    plan = current_pins(runner, state, actor, source)
    now = time.time()
    save('readiness.json', {'timestamp': timestamp(), 'online': True,
        'heartbeat_age_seconds': now - session['heartbeat_at'],
        'session_remaining_seconds': session['expires_at'] - now, 'plan_sha256': plan.plan_sha256,
        'selection_sha256': state['selection_sha256'], 'wave_id': state['wave_id'],
        'source_and_worktrees_unchanged': True, 'model_calls': 0, 'credential_copy': False})
    print(json.dumps({'ready': True, 'plan_sha256': plan.plan_sha256, 'model_calls': 0}))


def error_code(exc):
    if isinstance(exc, AgentSyncError) and re.fullmatch('[a-z_]{1,64}', exc.code):
        return exc.code
    return {'AssertionError': 'precondition_failed', 'TimeoutError': 'execution_timeout',
            'FileExistsError': 'attempt_already_reserved'}.get(type(exc).__name__, 'operation_failed')


def attempt_summary(attempt):
    fields = asdict(attempt)
    return {key: fields[key] for key in ATTEMPT_FIELDS}


async def execute_once(expected):
    runtime()
    core, runner = core_and_relay()
    state, source = pinned_state(core)
    assert expected == state['plan_sha256'] == load('readiness.json')['plan_sha256']
    assert not (BASE / 'attempt.json').exists()
    private_directory(HOME)
    require_mount()
    save('execute-entry.json', {'timestamp': timestamp(), 'plan_sha256': expected, 'retry_allowed': False})
    authorizations = 0

    async def approve(review):
        nonlocal authorizations
        if review.plan_sha256 != expected or authorizations:
            return None
        authorizations += 1
        return LaunchConsent('owner-host-coordinator-b455d9b-once', expected, time.time() + 120, True, True)

    actor = actor_for(state, approve)
    preview = current_pins(runner, state, actor, source)
    auth_file = HOME / 'auth.json'
    auth_info = auth_file.lstat()
    assert stat.S_ISREG(auth_info.st_mode) and stat.S_IMODE(auth_info.st_mode) == 0o600 and auth_info.st_uid == 0
    assert {entry.name for entry in HOME.iterdir()} == {'auth.json'}
    configs = [value for i, arg in enumerate(preview.plan.argv) if arg == '-c'
               for value in ('-c', preview.plan.argv[i + 1])]
    route = subprocess.run([str(EXE), *configs, 'login', 'status'], cwd=preview.plan.cwd,
                           env=preview.plan.child_env(), capture_output=True, timeout=20)
    recognized = route.returncode == 0 and (route.stdout + route.stderr).strip() == b'Logged in using ChatGPT'
    save('auth-route.json', {'recognized_chatgpt_route': recognized, 'exit_code': route.returncode, 'model_calls': 0})
    assert recognized
    save('attempt.json', {'timestamp': timestamp(), 'plan_sha256': expected, 'wave_id': state['wave_id'],
        'single_attempt_consumed': True, 'retry_allowed': False})
    capture = PrivateCapture(preview.plan, BASE / 'private-capture')
    queue = RelayQueue(runner)
    outcome, failure = None, None
    try:
        with capture:
            def receiving_actor(receiver, workspace):
                assert receiver == handoff_request(state).receiver
                assert (workspace.task_id, workspace.task_version, workspace.worktree_id,
                        workspace.head_sha, str(workspace.path)) == (
                    state['task']['id'], state['task']['version'], state['tree']['worktree_id'],
                    state['tree']['head_sha'], state['tree']['path'])
                return actor
            # Reopen the coordinator with its own saved source/task/tree receipts.
            # No source pin is manually forwarded and no producer impersonation occurs.
            outcome = await coordinator(core).run(handoff_request(state),
                admin_token=state['owner']['admin_token'], build_receiver=receiving_actor,
                wait_seconds=0, budget=EXECUTION_BUDGET)
    except Exception as exc:
        failure = error_code(exc)
    finally:
        await capture.retain()
        attempts = runner.owner_inspect_attempts(state['owner']['project_id'], state['task']['id'],
                                                 admin_token=state['owner']['admin_token'])
        record = {'timestamp': timestamp(), 'error_code': failure, 'captured_start_invoked': capture.entered,
            'exact_authorizations': authorizations, 'attempts': [attempt_summary(item) for item in attempts],
            'queue_outcome': None if outcome is None else asdict(outcome)}
        save('dispatch-observation.json', record)
    runtime()
    pinned_state(core)
    observed = runner.observe_codex_proposal_review(actor, state['task']['id'])
    saved_wave = queue.observe(actor, wave_request(state))
    pending = queue.owner_observe_pending(state['owner']['project_id'], admin_token=state['owner']['admin_token'])
    save('result.json', {'timestamp': timestamp(), 'plan_sha256': expected, 'wave_id': state['wave_id'],
        'queue': asdict(saved_wave), 'queue_pending': pending is not None, 'review': asdict(observed),
        'captured_start_invoked': capture.entered, 'error_code': failure, 'verification': 'unverified'})
    assert failure is None and capture.entered and authorizations == 1 and pending is None
    assert observed.state == 'submitted_for_review' and saved_wave.state == 'resolved'
    assert outcome is not None and outcome.state == 'resolved'
    assert outcome.task_id == state['task']['id'] and outcome.pin.context_id == PRODUCER_CONTEXT
    save('coordinator-result.json', {'timestamp': timestamp(), 'outcome': asdict(outcome),
        'resumed_from_saved_reference': True, 'exact_plan_authorizations': authorizations,
        'source_result_sha256': PRODUCER_RESULT_SHA, 'verification': 'unverified'})
    print(json.dumps({'submitted': True, 'verification': 'unverified'}))


def main():
    os.umask(0o077)
    assert Path(__file__).resolve() == BASE / 'pilot.py'
    private_directory(BASE)
    args = sys.argv[1:]
    if args == ['prepare']:
        prepare()
    elif len(args) == 2 and re.fullmatch('[0-9a-f]{64}', args[1]):
        if args[0] == 'ready-once':
            ready_once(args[1])
        elif args[0] == 'execute-once':
            asyncio.run(execute_once(args[1]))
        else:
            raise SystemExit('Use prepare or ready-once / execute-once with the reviewed plan SHA-256.')
    else:
        raise SystemExit('Use prepare or ready-once / execute-once with the reviewed plan SHA-256.')


if __name__ == '__main__':
    try:
        main()
    except Exception as exc:
        print(json.dumps({'error_code': error_code(exc), 'retry_allowed': False}))
        raise SystemExit(1) from None
