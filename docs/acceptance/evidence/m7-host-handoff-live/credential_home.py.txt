"""Mount and clean only the reviewed pilot's dedicated RAM-backed home."""

import json
import os
import stat
import subprocess
import sys
from datetime import UTC, datetime
from pathlib import Path

BASE = Path('/root/amadeus-host-handoff-live-20261113')
HOME = BASE / 'private-home'
LABEL = 'amadeus-host-handoff-auth'
ATTEMPT_FIELDS = (
    'leg', 'state', 'client', 'status', 'exit_code', 'timed_out', 'parent_ended', 'run_sha256',
    'session_sha256', 'error_code', 'admission', 'adapter_elapsed_ms', 'reported_input_tokens',
    'reported_output_tokens', 'agent_sha256', 'declared_client_version', 'protocol_session_sha256',
    'protocol_error_reason', 'stdout_bytes', 'stderr_bytes', 'stdout_sha256', 'stderr_sha256',
    'stdout_truncated', 'stderr_truncated',
)


def mounts():
    entries = [row.split() for row in Path('/proc/mounts').read_text().splitlines()]
    return [entry for entry in entries if len(entry) >= 4 and entry[1] == str(HOME)]


def require_mount():
    table = mounts()
    assert len(table) == 1 and table[0][0] == LABEL and table[0][2] == 'tmpfs'
    assert {'rw', 'nosuid', 'nodev', 'mode=700'} <= set(table[0][3].split(','))


def save(name, detail):
    with (BASE / name).open('x', encoding='utf-8') as out:
        json.dump({'timestamp': datetime.now(UTC).isoformat(), **detail}, out, indent=2)
        out.write('\n')


def require_stopped():
    if not (BASE / 'attempt.json').exists():
        return
    observed = json.loads((BASE / 'dispatch-observation.json').read_text(encoding='utf-8'))
    attempts = observed['attempts']
    assert type(attempts) is list and len(attempts) == 2
    assert all(type(attempt) is dict and set(attempt) == set(ATTEMPT_FIELDS) for attempt in attempts)
    first, second = attempts
    assert first['leg'] == 'first' and second['leg'] == 'second'
    if observed['captured_start_invoked'] is False:
        assert type(observed['exact_authorizations']) is int and observed['exact_authorizations'] == 0
        for attempt in attempts:
            assert attempt['state'] == 'not_dispatched'
            assert all(value is None for key, value in attempt.items() if key not in {'leg', 'state'})
        return
    assert observed['captured_start_invoked'] is True
    assert first['state'] == 'terminal_recorded' and first['parent_ended'] is True
    assert second['state'] == 'not_dispatched'


def main():
    os.umask(0o077)
    assert Path(__file__).resolve() == BASE / 'credential_home.py'
    for path in (BASE, HOME):
        info = path.lstat()
        assert stat.S_ISDIR(info.st_mode) and info.st_uid == 0 and stat.S_IMODE(info.st_mode) == 0o700
    if sys.argv[1:] == ['mount']:
        assert (BASE / 'plan.json').is_file() and not mounts() and not any(HOME.iterdir())
        assert not any((BASE / name).exists() for name in ('credential-home.json', 'execute-entry.json', 'attempt.json'))
        subprocess.run(['/bin/mount', '-t', 'tmpfs', '-o', 'size=64m,mode=700,nodev,nosuid',
                        LABEL, str(HOME)], check=True, capture_output=True, timeout=10)
        require_mount()
        save('credential-home.json', {'temporary_storage': 'tmpfs', 'mode': '0700', 'credentials_copied': False})
        print('Dedicated RAM-backed home ready.')
    elif sys.argv[1:] == ['cleanup']:
        require_stopped()
        require_mount()
        auth = HOME / 'auth.json'
        if os.path.lexists(auth):
            entry = auth.lstat()
            assert stat.S_ISREG(entry.st_mode) and entry.st_uid == 0
            auth.unlink()
        assert not os.path.lexists(auth)
        subprocess.run(['/bin/umount', str(HOME)], check=True, capture_output=True, timeout=10)
        assert not mounts() and not any(HOME.iterdir())
        save('auth-cleanup.json', {'temporary_auth_absent': True, 'home_unmounted': True,
                                   'private_home_empty': True})
        print('Temporary authentication removed and dedicated RAM filesystem unmounted.')
    else:
        raise SystemExit('Choose mount or cleanup.')


if __name__ == '__main__':
    try:
        main()
    except Exception:
        raise SystemExit('Credential home operation stopped. Inspect privately. Do not retry the pilot.') from None
