"""Read one exact receiver request, or authorize its previously inspected bytes."""
import hashlib
import json
import re
import shlex
import stat
import sys
from pathlib import Path

import paramiko

BASE = Path(__file__).resolve().parent
REMOTE = '/root/amadeus-interrupted-a117-r2-20261116'
PYTHON = '/root/amadeus-demo-a117-sdist-r1/installed/bin/python'


def main():
    args = sys.argv[1:]
    assert args == ['read'] or len(args) == 3 and args[0] == 'approve'
    if args[0] == 'approve':
        assert all(re.fullmatch('[a-f0-9]{64}', value) for value in args[1:])
    client = paramiko.SSHClient()
    client.load_host_keys('D:/agentsync-debug/vm/known_hosts')
    client.set_missing_host_key_policy(paramiko.RejectPolicy())
    client.connect('127.0.0.1', port=22222, username='root',
                   key_filename='D:/agentsync-debug/vm/debug_ed25519',
                   look_for_keys=False, allow_agent=False, timeout=10)
    try:
        with client.open_sftp() as sftp:
            info = sftp.lstat(REMOTE + '/receiver-request.json')
            assert stat.S_ISREG(info.st_mode) and stat.S_IMODE(info.st_mode) == 0o600
            assert 0 < info.st_size <= 262144
            with sftp.open(REMOTE + '/receiver-request.json', 'rb') as stream:
                raw = stream.read(262144)
        request = json.loads(raw)
        digest = hashlib.sha256(raw).hexdigest()
        if args[0] == 'read':
            with (BASE / 'receiver-request.json').open('xb') as stream:
                stream.write(raw)
            print(json.dumps({'request_sha256': digest, 'request': request}))
            return
        assert raw == (BASE / 'receiver-request.json').read_bytes()
        assert digest == args[1] and request['plan_sha256'] == args[2]
        approved = json.loads((BASE / 'continuation-plan.json').read_bytes())
        original = json.loads((BASE / 'plan.json').read_bytes())['producer_plan']
        assert request['stage'] == 'receiver'
        assert request['continuation_plan_sha256'] == approved['plan_sha256']
        assert request['authorization_sha256'] == approved['authorization_sha256']
        payload = request['payload']
        for field in ('environment', 'timeout_seconds', 'output_limit', 'binary_sha256', 'client_version'):
            assert payload[field] == original[field]
        cwd = payload['cwd']
        assert re.fullmatch(re.escape(REMOTE) + r'/core/managed-git/worktrees/[a-f0-9-]{36}', cwd)
        assert payload['argv'] == [arg.replace(original['cwd'], cwd) for arg in original['argv']]
        assert payload['stdin_text'].startswith('Implement add(a, b) in add.py so that it returns a + b for two integers. Finish the pinned unfinished function. Do not use tools, read or modify files, or run code.\n')
        contract, _ = json.JSONDecoder().raw_decode(payload['stdin_text'].split('Host contract: ', 1)[1])
        assert contract['allowed_paths'] == ['add.py'] and contract['submit_for_review'] is True
        assert contract['mode'] == 'proposal_continuation' and contract['base_head'] == approved['plan']['base_head']
        source = approved['plan']['source']
        assert contract['continuation'] == source | {'source_task_version': source['source_task_version'] + 1}
        probe = '''import json,sys,time
from pathlib import Path
root=Path('/root/amadeus-interrupted-a117-r2-20261116');sys.path.insert(0,str(root))
import cleanup,live
request=live.read('receiver-request.json')
assert live.sha(root/'receiver-request.json')==sys.argv[1]
assert request['stage']=='receiver' and request['plan_sha256']==sys.argv[2]
assert request['expires_at']>time.time()+2
assert cleanup.validate_authorization()[1]==request['authorization_sha256']
decision={'stage':'receiver','plan_sha256':request['plan_sha256'],'request_sha256':sys.argv[1],'expires_at':request['expires_at']}
cleanup.save('receiver-decision.json',decision)
print(json.dumps(decision))
'''
        command = PYTHON + ' -I -B - ' + shlex.quote(args[1]) + ' ' + shlex.quote(args[2])
        stdin, stdout, stderr = client.exec_command(command, timeout=10)
        stdin.write(probe); stdin.flush(); stdin.channel.shutdown_write()
        output, errors = stdout.read(8193), stderr.read(8193)
        code = stdout.channel.recv_exit_status()
        assert code == 0 and not errors and len(output) <= 8192
        decision = json.loads(output)
        with (BASE / 'receiver-decision.json').open('x', encoding='utf-8') as stream:
            json.dump(decision, stream, indent=2)
        print(json.dumps(decision))
    finally:
        client.close()


if __name__ == '__main__':
    main()
