def sha(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def write_json(path, record):
    with path.open("x", encoding="utf-8", newline="\n") as stream:
        json.dump(record, stream, indent=2)
        stream.write("\n")


def verify_inputs():
    for path, expected in IDENTITIES.items():
        if sha(path) != expected:
            raise ValueError(f"Pinned input changed: {path.name}")
    manifest = json.loads((BASE / "frozen-inputs.json").read_bytes())
    members = json.loads((BASE / "runtime-members.json").read_bytes())
    expected = {item["path"]: item for item in manifest["files"]}
    actual = {file.relative_to(SOURCE).as_posix(): file for file in SOURCE.rglob("*") if file.is_file()}
    if (
        len(expected) != len(manifest["files"]) or len(expected) != 2382
        or expected.keys() != actual.keys() or len(members) != 97
        or manifest["test_source_commit"] != TEST_COMMIT
        or manifest["product_runtime_commit"] != PRODUCT_COMMIT
    ):
        raise ValueError("Frozen input inventory or source identity changed")
    for name, entry in expected.items():
        file = actual[name]
        if file.resolve() != file or file.stat().st_size != entry["bytes"] or sha(file) != entry["sha256"]:
            raise ValueError(f"Frozen file changed: {name}")
    if any(expected[f"src/{name}"]["sha256"] != digest for name, digest in members.items()):
        raise ValueError("Frozen fixture source and pinned product inventory disagree")
    return {"frozen_files": len(expected), "expected_runtime_members": len(members)}


def isolate_environment():
    for name in list(os.environ):
        upper = name.upper()
        if upper.startswith((
            "AGENTSYNC_", "ANTHROPIC_", "CLAUDE_", "CODEX_", "CURSOR_", "DEEPSEEK_", "DOCKER_",
            "GEMINI_",
            "GIT_", "GOOGLE_", "OPENAI_", "PIP_", "PYTEST_", "PYTHON", "UV_",
        )) or upper.endswith(("API_KEY", "ACCESS_TOKEN", "REFRESH_TOKEN")):
            os.environ.pop(name, None)
    os.environ.update(
        HOME=str(HOME), USERPROFILE=str(HOME), CODEX_HOME=str(HOME),
        DOCKER_CONFIG=str(HOME / ".docker"), DOCKER_HOST="unix:///var/run/docker.sock",
        XDG_CONFIG_HOME=str(HOME / "config"), XDG_CACHE_HOME=str(HOME / "cache"),
        TEMP=str(SCRATCH), TMP=str(SCRATCH), TMPDIR=str(SCRATCH),
        PYTHONDONTWRITEBYTECODE="1", PYTHONNOUSERSITE="1",
        GIT_CONFIG_NOSYSTEM="1", GIT_CONFIG_GLOBAL=os.devnull, GIT_TERMINAL_PROMPT="0",
        GCM_INTERACTIVE="Never", PIP_CONFIG_FILE=os.devnull, PIP_NO_INDEX="1",
        RUFF_CACHE_DIR=str(SCRATCH / "ruff"), MYPY_CACHE_DIR=str(SCRATCH / "mypy"),
    )

def docker(*arguments):
    for executable in ("/usr/bin/docker", "/usr/local/bin/docker"):
        if os.access(executable, os.X_OK):
            return subprocess.check_output([executable, *arguments], text=True, timeout=30)
    raise FileNotFoundError("Local Docker CLI is unavailable")


def docker_evidence(root):
    """Require actual verifier and paired proposal observations; hash all fixture reports."""
    verifier = list((root / "verifier").glob("*/test_actual_docker_checks_all_external_cases.json"))
    if len(verifier) != 1:
        raise ValueError("Expected one actual isolation verifier observation")
    direct = json.loads(verifier[0].read_bytes())
    if direct["image_id"] != DOCKER_IMAGE or direct["cleanup_confirmed"] is not True:
        raise ValueError("Verifier observation has wrong image or unconfirmed cleanup")
    for name, verdict in (("verified", "verified"), ("wrong-result", "unverified")):
        pilot = json.loads((root / "proposal_pipeline" / f"{name}.json").read_bytes())
        execution = pilot["execution"]["execution"]["result"]
        if (
            pilot["model_calls"] != 0 or pilot["task_verification"] != verdict
            or pilot["container_absent"] is not True or execution["cleanup_confirmed"] is not True
            or execution["backend"] != "local-linux-docker"
            or execution["image_sha256"] != DOCKER_IMAGE
        ):
            raise ValueError(f"Invalid actual Docker proposal observation: {name}")
    return {
        path.relative_to(RUN).as_posix(): sha(path) for path in sorted(root.rglob("*.json"))
    }


def run():
    if (
        sys.platform != "linux" or sys.version_info[:3] != (3, 14, 7)
        or Path(sys.executable) != BASE_PYTHON or Path(sys.prefix) != BASE_PYTHON.parent.parent
        or not sys.flags.isolated or not sys.dont_write_bytecode or not __debug__
    ):
        raise SystemExit(f"Use the pinned Linux 3.14.7 interpreter with -I -B: {BASE_PYTHON}")
    if not DOCKER_IMAGE:
        raise SystemExit("The preinstalled Docker image has not been pinned.")
    if any(os.path.lexists(path) for path in (RUN, INSTALL, SCRATCH, HOME)):
        raise SystemExit("Attempt already exists. Preserve it without retrying.")
    os.umask(0o077)
    RUN.mkdir()
    (RUN / "logs").mkdir()
    SCRATCH.mkdir()
    HOME.mkdir()
    isolate_environment()
    result = {
        "scope": "Linux 3.14, one full installed a117 preview sdist suite with actual local Docker",
        "test_source_commit": TEST_COMMIT,
        "product_runtime_commit": PRODUCT_COMMIT,
        "version": VERSION,
        "sdist_sha256": ARCHIVE_SHA,
        "fixture_snapshot_sha256": IDENTITIES[BASE / "frozen-inputs.tar"],
        "driver_sha256": sha(Path(__file__)),
        "controller_pid": os.getpid(),
        "controller_python": sys.version,
        "boot_id": Path("/proc/sys/kernel/random/boot_id").read_text().strip(),
        "required_docker_image_id": DOCKER_IMAGE,
        "model_calls": 0,
        "started_at": datetime.now(UTC).isoformat(),
        "status": "unknown",
        "formal_release": "BLOCKED",
        "automatic_retry": False,
        "checks": [],
    }
    write_json(RUN / "attempt.json", result)

    try:
        result["input_before"] = verify_inputs()
        result["docker_before"] = {
            "containers": docker("ps", "-aq", "--no-trunc").splitlines(),
            "image_id": docker("image", "inspect", "--format", "{{.Id}}", DOCKER_IMAGE).strip(),
        }
        if result["docker_before"] != {"image_id": DOCKER_IMAGE, "containers": []}:
            raise RuntimeError("Docker preflight needs the pinned image and an empty daemon")
        sys.path.insert(0, str(SOURCE))
        from tools import release_acceptance as collector

        project = tomllib.loads((SOURCE / "pyproject.toml").read_text(encoding="utf-8"))
        packages = collector._prepare_offline_packages(project, WHEELHOUSE, SCRATCH, RUN)
        python = str(INSTALL / "bin" / "python")
        origin_command = [
            python, "-I", "-B", "-c", ORIGIN_PROBE,
            str(INSTALL), str(ARCHIVE), str(BASE / "runtime-members.json"), ARCHIVE_SHA, VERSION,
        ]

        def execute(name, command, timeout=900):
            outcome = collector.execute(
                name, command, RUN, RUN, SCRATCH,
                pytest_timeout=timeout, offline_wheelhouse=packages.wheelhouse,
            )
            result["checks"].append(outcome)
            return outcome

        def require(name, command):
            if execute(name, command)["status"] != "pass":
                raise RuntimeError(f"{name} failed")

        require("sdist_venv", [sys.executable, "-I", "-B", "-m", "venv", str(INSTALL)])
        require("sdist_offline_dependencies", collector._offline_install(
            python, packages.wheelhouse, "--only-binary=:all:", *packages.backend, *packages.application,
        ))
        require("sdist_install", collector._offline_install(
            python, packages.wheelhouse, "--no-build-isolation", "--no-deps", "--", str(ARCHIVE),
        ))
        require("sdist_pip_check", [python, "-I", "-B", "-m", "pip", "check"])
        require("sdist_origin_before", origin_command)

        raw_junit = SCRATCH / "sdist-junit.xml"
        observations = RUN / "observations"
        for group in EVIDENCE_GROUPS:
            path = observations / group.lower()
            path.mkdir(parents=True)
            os.environ[f"AGENTSYNC_{group}_EVIDENCE_DIR"] = str(path)
        os.environ.update(AGENTSYNC_DOCKER_TEST_IMAGE=DOCKER_IMAGE, AGENTSYNC_REQUIRE_DOCKER_TESTS="1")
        try:
            tested = execute("sdist_pytest", [
                python, "-I", "-B", "-m", "pytest", str(SOURCE / "tests"),
                "--junitxml", str(raw_junit), "--basetemp", str(SCRATCH / "p"),
                "-o", f"cache_dir={SCRATCH / 'pytest-cache'}",
            ], timeout=7200)
        finally:
            origin_after = execute("sdist_origin_after", origin_command)
            if raw_junit.is_file():
                junit = RUN / "sdist-junit.xml"
                redactions = collector.persist_junit(raw_junit, junit)
                cases = list(ET.parse(junit).iter("testcase"))
                result["junit"] = {
                    "path": junit.name, "sha256": sha(junit), "credential_redactions": redactions,
                    "cases": len(cases), **{
                        field: sum(case.find(field) is not None for case in cases)
                        for field in ("failure", "error", "skipped")
                    },
                }
        if (
            tested["status"] != "pass" or origin_after["status"] != "pass"
            or result.get("junit", {}).get("cases", 0) == 0
            or result["junit"]["failure"] or result["junit"]["error"]
        ):
            raise RuntimeError("Full installed-sdist pytest or its evidence checks failed")
        result["fixture_observations_sha256"] = docker_evidence(observations)
        result["status"] = "pass"
    except Exception as exc:
        result["status"] = "fail"
        result["error_type"] = type(exc).__name__
    finally:
        if "docker_before" in result:
            try:
                result["docker_after"] = {"containers": docker("ps", "-aq", "--no-trunc").splitlines()}
                if result["docker_after"]["containers"]:
                    result["status"] = "fail"
            except (OSError, subprocess.SubprocessError) as exc:
                result["status"] = "fail"
                result["docker_after_error_type"] = type(exc).__name__
        if "input_before" in result:
            try:
                result["input_after"] = verify_inputs()
            except (OSError, ValueError, KeyError) as exc:
                result["status"] = "fail"
                result["input_after_error_type"] = type(exc).__name__
        result["finished_at"] = datetime.now(UTC).isoformat()
        write_json(RUN / "result.json", result)
        print(f"{result['status']}: {RUN / 'result.json'}", flush=True)
    return 0 if result["status"] == "pass" else 1


if __name__ == "__main__":
    raise SystemExit(run())
