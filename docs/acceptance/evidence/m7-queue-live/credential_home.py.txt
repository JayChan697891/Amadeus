"""Manage only this pilot's RAM-backed temporary credential directory."""
from datetime import UTC, datetime
import json
import os
from pathlib import Path
import stat
import subprocess
import sys

BASE=Path('/root/amadeus-queued-heartbeat-live-20261112')
HOME=BASE/'private-home'
os.umask(0o077)
assert not BASE.is_symlink() and not HOME.is_symlink()
assert BASE.stat().st_uid==HOME.stat().st_uid==0
assert stat.S_IMODE(BASE.stat().st_mode)==stat.S_IMODE(HOME.stat().st_mode)==0o700
def mounts():
    return [row.split() for row in Path('/proc/mounts').read_text().splitlines() if row.split()[1]==str(HOME)]
def save(name,value):
    with (BASE/name).open('x') as out:json.dump(value,out,indent=2)

def cleanup_supported(result):
    attempts=result['attempts']
    assert len(attempts)==2 and [item['leg'] for item in attempts]==['first','second']
    if result.get('captured_start_invoked') is False:
        for item in attempts:
            assert item['state']=='not_dispatched'
            assert all(value is None for key,value in item.items()
                       if key not in {'leg','state','billing_verified','cost_usd','verification'})
        return
    assert result.get('captured_start_invoked') is True
    assert attempts[0]['state']=='terminal_recorded' and attempts[0]['parent_ended'] is True
    assert attempts[1]['state']=='not_dispatched'

if sys.argv[1:] == ['mount']:
    assert not mounts() and not any(HOME.iterdir())
    assert not (BASE/'execute-entry.json').exists()
    subprocess.run(['/bin/mount','-t','tmpfs','-o','size=64m,mode=700,nodev,nosuid','amadeus-live-auth',str(HOME)],check=True,capture_output=True,timeout=10)
    table=mounts()
    assert len(table)==1 and table[0][0]=='amadeus-live-auth' and table[0][2]=='tmpfs'
    save('credential-home.json',{'timestamp':datetime.now(UTC).isoformat(),'temporary_storage':'tmpfs','mode':'0700','credentials_copied':False})
    print('Dedicated RAM-backed home ready; no credential copied by this helper.')
elif sys.argv[1:] == ['cleanup']:
    if (BASE/'attempt.json').exists():
        result=json.loads((BASE/'result.json').read_text())
        cleanup_supported(result)
    table=mounts()
    assert len(table)==1 and table[0][0]=='amadeus-live-auth' and table[0][2]=='tmpfs'
    auth=HOME/'auth.json'
    if auth.exists():
        item=auth.lstat()
        assert stat.S_ISREG(item.st_mode) and item.st_uid==0
        auth.unlink()
    assert not auth.exists()
    subprocess.run(['/bin/umount',str(HOME)],check=True,capture_output=True,timeout=10)
    assert not mounts() and not any(HOME.iterdir())
    save('auth-cleanup.json',{'timestamp':datetime.now(UTC).isoformat(),'temporary_auth_absent':True,'home_unmounted':True,'private_home_empty':True})
    print('Temporary authentication removed and dedicated RAM filesystem unmounted.')
else:
    raise SystemExit('Choose mount or cleanup')
