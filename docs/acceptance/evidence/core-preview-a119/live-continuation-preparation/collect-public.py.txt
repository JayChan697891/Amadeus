"""Post-run collection recipe; original private diagnostics remain on D:."""
import hashlib
import json
import re
import shutil
import xml.etree.ElementTree as ET
from pathlib import Path

repo = Path('F:/syuc/amadeus-m6-single')
out = repo / 'docs/acceptance/evidence/core-preview-a119/live-continuation-preparation'
root = Path('D:/agentsync-debug/strategy-demo/a119-solution-review-20261116')
helpers = Path('D:/agentsync-debug/strategy-demo/interrupted-a119-solution-20261116')
tests = Path(str(helpers) + '-offline')
sha = lambda b: hashlib.sha256(b).hexdigest()
records = []
def copy(path, name, redact=False):
    data = path.read_bytes()
    public = data
    if redact:
        text = data.decode('utf-8-sig')
        pattern = r"(['\"](?:admin_token|session_token|lease_token|join_token)['\"]\s*:\s*)(['\"])([^'\"\r\n]+)\2"
        text, count = re.subn(pattern, lambda match: match[1] + match[2] + '[REDACTED synthetic local credential]' + match[2], text)
        public = text.encode('utf-8')
    else:
        count = 0
    target = out / name
    with target.open('xb') as stream:
        stream.write(public)
    records.append({'file': name, 'bytes': len(public), 'sha256': sha(public), 'source_sha256': sha(data), 'redacted_fields': count})

for name in ('check-destinations.sh', 'check-destinations.log', 'check-preparation.sh', 'check-preparation.log'):
    copy(root / name, name)
for name in ('prepared-inputs.json', 'bound-helpers.json', 'live-authorization.template.json'):
    copy(helpers / name, name)
for name in ('final2.log', 'final2.xml', 'final2.exit', 'walkthrough-r3.log', 'walkthrough-r3.xml', 'walkthrough-r3.exit', 'walkthrough-r2.log', 'walkthrough-r2.xml', 'walkthrough-r2.exit', 'walkthrough.exit'):
    copy(tests / name, name)
for name in ('test_preparation.py', 'test_walkthrough.py', 'walkthrough-first.py.txt', 'walkthrough-r2.py.txt', 'walkthrough-r2-verify.py.txt'):
    copy(tests / name, name if name.endswith('.txt') else name + '.txt')
for name in ('walkthrough.log', 'walkthrough.xml'):
    copy(tests / name, 'redacted-' + name, redact=True)
for name, count in (('final2.xml', 11), ('walkthrough-r3.xml', 1)):
    suite = ET.fromstring((out / name).read_bytes()).find('testsuite')
    assert suite is not None and int(suite.attrib['tests']) == count
    assert all(int(suite.attrib[key]) == 0 for key in ('errors', 'failures', 'skipped'))
for name in ('walkthrough-r2.xml', 'redacted-walkthrough.xml'):
    suite = ET.fromstring((out / name).read_bytes()).find('testsuite')
    assert suite is not None and int(suite.attrib['failures']) == 1
copy(Path(__file__), 'collect-public.py.txt')
with (out / 'collection.json').open('x', encoding='utf-8') as stream:
    json.dump({'kind': 'post-run public collection, not a pre-run command receipt', 'files': records}, stream, indent=2)
    stream.write('\n')
print(json.dumps({'collected': len(records), 'redacted_fields': sum(r['redacted_fields'] for r in records), 'final_gate_tests': 11, 'final_walkthrough_tests': 1, 'preserved_failures': 2}))
