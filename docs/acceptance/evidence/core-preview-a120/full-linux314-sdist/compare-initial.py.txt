import hashlib,json,xml.etree.ElementTree as ET
from pathlib import Path

base=Path('D:/agentsync-debug/m120sdistl');run=base/'harvest'
reference=Path('F:/syuc/amadeus-m6-single/docs/acceptance/evidence/core-preview-a120/full-linux314')
def sha(p):return hashlib.sha256(p.read_bytes()).hexdigest()
def cases(path):
    rows={}
    for node in ET.parse(path).iter('testcase'):
        key=node.attrib['classname']+'::'+node.attrib['name'];assert key not in rows
        outcome=next((name for name in ('failure','error','skipped') if node.find(name) is not None),'passed')
        rows[key]={'outcome':outcome}
        if outcome!='passed':rows[key]['message']=node.find(outcome).attrib.get('message','')
    return rows
result=json.loads((run/'result.json').read_bytes());assert result['finished_at']
current=cases(run/'sdist-junit.xml');wheel=cases(reference/'wheel-junit.xml')
counts={s:sum(row['outcome']==s for row in current.values()) for s in ('passed','skipped','failure','error')}
checks=[]
for check in result['checks']:
    receipt=json.loads((run/'logs'/f"{check['id']}.process.json").read_bytes())
    assert receipt['phase']=='finished' and receipt['outcome']=='exited' and receipt['result']==check
    assert check['status']=='pass' and check['exit_code']==0 and not check['log_truncated']
    assert sha(run/check['log'])==check['log_sha256']
    checks.append({'check':check['id'],'exit_code':check['exit_code'],'log_sha256':check['log_sha256']})
assert sha(run/'sdist-junit.xml')==result['junit']['sha256']
comparison={'unique_cases':len(current),'counts':counts,'wheel_case_identity_equal':current.keys()==wheel.keys(),'removed_cases':sorted(wheel.keys()-current.keys()),'added_cases':sorted(current.keys()-wheel.keys()),'outcome_differences':{k:{'wheel':wheel[k],'sdist':current[k]} for k in sorted(current.keys()&wheel.keys()) if current[k]['outcome']!=wheel[k]['outcome']},'skips':{k:v for k,v in current.items() if v['outcome']=='skipped'},'checks':checks}
with (base/'case-diff.json').open('x',encoding='utf8') as f:json.dump(comparison,f,indent=2);f.write('\n')
fields=set(json.loads((reference/'docker-diff.json').read_bytes())['projection_fields'])|{'same_core','source_task_status','receiver_task_status','receiver_task_verification','publication_verification'}
def projection(value,prefix=''):
    out={}
    if isinstance(value,dict):
        for key,item in value.items():
            path=prefix+'/'+key
            if key in fields or key.endswith(('_case_count','_exit_code')):out[path]=item
            elif key.endswith('_json') and isinstance(item,str):out.update(projection(json.loads(item),path))
            else:out.update(projection(item,path))
    elif isinstance(value,list):
        for index,item in enumerate(value):out.update(projection(item,prefix+'/'+str(index)))
    return out
old={p.relative_to(reference/'observations').as_posix():p for p in (reference/'observations').rglob('*.json')}
new={p.relative_to(run/'observations').as_posix():p for p in (run/'observations').rglob('*.json')}
differences=[]
for name in sorted(old.keys()&new.keys()):
    a,b=projection(json.loads(old[name].read_bytes())),projection(json.loads(new[name].read_bytes()))
    for key in sorted(a.keys()|b.keys()):
        if a.get(key)!=b.get(key):differences.append({'file':name,'field':key,'wheel':a.get(key),'sdist':b.get(key)})
docker={'wheel_observations':len(old),'sdist_observations':len(new),'removed':sorted(old.keys()-new.keys()),'added':sorted(new.keys()-old.keys()),'differences':differences,'projection_fields':sorted(fields),'projection_suffixes':['_case_count','_exit_code'],'scope':'Selected semantic outcome/isolation/cleanup comparison; not byte equality or new real-client proof.'}
with (base/'docker-diff.json').open('x',encoding='utf8') as f:json.dump(docker,f,indent=2);f.write('\n')
print(json.dumps({k:v for k,v in comparison.items() if k not in ('skips','checks')}))
print(json.dumps({k:v for k,v in docker.items() if k not in ('projection_fields','differences')}))
print(json.dumps({'docker_differences':differences}))
assert counts=={'passed':2580,'skipped':7,'failure':0,'error':0} and current.keys()==wheel.keys() and not comparison['outcome_differences']
assert len(checks)==4 and len(old)==len(new)==36 and not docker['added'] and not docker['removed'] and not differences
