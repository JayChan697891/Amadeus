from pathlib import Path
import hashlib,json,sys
import agentsync,pytest
base=Path(__file__).resolve().parent
config=json.loads((base/'linux-config.json').read_text())
site=Path(agentsync.__file__).resolve().parent.parent
assert site.is_relative_to(base/'venv')
assert hashlib.sha256((base/'agentsync_local-0.3.0a109-py3-none-any.whl').read_bytes()).hexdigest()==config['wheel_sha256']
def verify():
 for rel,expected in config['module_hashes'].items():assert hashlib.sha256((site/rel).read_bytes()).hexdigest()==expected,rel
 for name,module in tuple(sys.modules.items()):
  if name=='agentsync' or name.startswith('agentsync.'):assert Path(module.__file__).resolve().is_relative_to(site/'agentsync'),name
verify()
code=pytest.main([str(base/'source/tests/integration/test_codex_proposal_review.py'),'-q','-p','no:cacheprovider','--basetemp='+str(base/'test-tmp'),'--junitxml='+str(base/'installed.xml')])
verify()
(base/'origin.json').write_text(json.dumps({'product_sha':config['product_sha'],'wheel_sha256':config['wheel_sha256'],'python':sys.version,'runtime_modules_matched_before_and_after':len(config['module_hashes']),'source_shadowing':False,'exit_code':code,'model_requests':0,'candidate_code_execution':False},indent=2))
raise SystemExit(code)
