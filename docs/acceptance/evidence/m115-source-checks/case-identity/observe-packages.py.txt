"""Observe existing package runs and preserve raw identities locally; never dispatch tests."""
import ctypes
from ctypes import wintypes
from datetime import datetime, timezone
import hashlib
import json
import msvcrt
import os
from pathlib import Path
import subprocess
import sys
import time

BASE = Path(__file__).resolve().parent
RUN = Path('D:/agentsync-debug/m115-r3/windows-312/release-20261114T035240Z-6756b7e7fa0d')
SCRATCH = Path('D:/agentsync-debug/t115u/.scratch-u_ak32ml')
HELPER = BASE / 'recover_identity.py'
kernel = ctypes.WinDLL('kernel32', use_last_error=True)
kernel.CreateFileW.argtypes = [wintypes.LPCWSTR, wintypes.DWORD, wintypes.DWORD, wintypes.LPVOID, wintypes.DWORD, wintypes.DWORD, wintypes.HANDLE]
kernel.CreateFileW.restype = wintypes.HANDLE
kernel.OpenProcess.argtypes = [wintypes.DWORD, wintypes.BOOL, wintypes.DWORD]
kernel.OpenProcess.restype = wintypes.HANDLE
kernel.WaitForSingleObject.argtypes = [wintypes.HANDLE, wintypes.DWORD]
kernel.WaitForSingleObject.restype = wintypes.DWORD
kernel.CloseHandle.argtypes = [wintypes.HANDLE]
kernel.CloseHandle.restype = wintypes.BOOL

def stamp():
    return datetime.now(timezone.utc).isoformat()

def save(name, value):
    with (BASE / name).open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')

command = "Get-CimInstance Win32_Process -Filter 'ProcessId = 4488' | Select-Object ProcessId,@{Name='CreationDate';Expression={$_.CreationDate.ToUniversalTime().ToString('o')}},CommandLine | ConvertTo-Json -Compress"
identity = json.loads(subprocess.check_output(['powershell.exe', '-NoProfile', '-NonInteractive', '-Command', command], text=True, timeout=15))
assert identity['ProcessId'] == 4488
assert datetime.fromisoformat(identity['CreationDate']).astimezone(timezone.utc).isoformat() == '2026-11-14T03:52:40.575725+00:00'
assert 'm115-r3' in identity['CommandLine'] and 'release_acceptance.py' in identity['CommandLine']
process = kernel.OpenProcess(0x100000, False, 4488)
assert process and kernel.WaitForSingleObject(process, 0) == 258
helper_sha = hashlib.sha256(HELPER.read_bytes()).hexdigest()
save('observer-attempt.json', {'pid': os.getpid(), 'started_at': stamp(), 'collector': identity,
                             'helper_sha256': helper_sha, 'tests_dispatched': False, 'automatic_retry': False})
deadline = time.monotonic() + 10800
try:
    for suite in ('wheel', 'sdist'):
        raw_path = SCRATCH / (suite + '-junit.xml')
        durable = RUN / (suite + '-junit.xml')
        receipt_path = RUN / 'logs' / (suite + '_pytest.process.json')
        held = None
        try:
            while True:
                if held is None:
                    # SHARE_READ | SHARE_WRITE | SHARE_DELETE: do not delay collector cleanup.
                    handle = kernel.CreateFileW(str(raw_path), 0x80000000, 7, None, 3, 0x80, None)
                    if handle != ctypes.c_void_p(-1).value:
                        try:
                            descriptor = msvcrt.open_osfhandle(handle, os.O_RDONLY | os.O_BINARY)
                        except BaseException:
                            kernel.CloseHandle(handle)
                            raise
                        held = os.fdopen(descriptor, 'rb')
                        save(suite + '-raw-handle.json', {'opened_at': stamp(), 'path': str(raw_path), 'share_read_write_delete': True})
                receipt = None
                if receipt_path.is_file():
                    try:
                        receipt = json.loads(receipt_path.read_text(encoding='utf-8'))
                    except json.JSONDecodeError:
                        pass  # A receipt write in progress is not a terminal result.
                if receipt and receipt.get('phase') == 'finished' and durable.is_file():
                    assert held is not None, 'Raw XML capture missing; preserve blocked state, never retry tests'
                    held.seek(0)
                    payload = held.read()
                    target = BASE / 'private' / (suite + '-junit.xml')
                    with target.open('xb') as stream:
                        stream.write(payload)
                    save(suite + '-raw-capture.json', {'captured_at': stamp(), 'sha256': hashlib.sha256(payload).hexdigest(),
                                                   'size': len(payload), 'receipt_exit_code': receipt['result']['exit_code'],
                                                   'storage': 'Restricted local only; never upload raw'})
                    assert receipt['result']['exit_code'] == 0, 'Actual suite failed; no identity acceptance'
                    break
                assert kernel.WaitForSingleObject(process, 0) == 258, 'Collector ended without complete capture; inspect originals'
                assert time.monotonic() < deadline, 'Observer deadline reached; no collector termination or retry'
                time.sleep(0.2)
        finally:
            if held is not None:
                held.close()
        assert hashlib.sha256(HELPER.read_bytes()).hexdigest() == helper_sha, 'Recovery helper changed during observation'
        started = stamp()
        with (BASE / (suite + '-recovery.log')).open('xb') as log:
            result = subprocess.run([sys.executable, '-I', '-B', str(HELPER), suite], stdout=log, stderr=subprocess.STDOUT, timeout=60)
        save(suite + '-recovery-process.json', {'started_at': started, 'finished_at': stamp(), 'exit_code': result.returncode,
                                             'helper_sha256': helper_sha, 'new_test_execution': False})
        assert result.returncode == 0, 'Recovery failed; retain original evidence, do not retry'
        print(json.dumps({'suite': suite, 'identity_recovery': 'finished', 'new_tests': False}), flush=True)
    save('observer-result.json', {'finished_at': stamp(), 'exit_code': 0, 'scope': 'Raw preservation and derived identities only; full report validation pending'})
except BaseException as exc:
    save('observer-result.json', {'finished_at': stamp(), 'exit_code': 1, 'error_type': type(exc).__name__, 'reason': str(exc), 'automatic_retry': False})
    raise
finally:
    kernel.CloseHandle(process)
