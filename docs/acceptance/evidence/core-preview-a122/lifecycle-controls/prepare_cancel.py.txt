"""Prepare one credential-free isolated fixture; no model launch."""
import hashlib
import json
import shlex
import stat
from pathlib import Path

import paramiko

base = Path(__file__).parent
source = Path('D:/agentsync-debug/m122-lifecycle-preparation')
remote = '/root/amadeus-lifecycle-a122-helpers'
root = '/root/amadeus-lifecycle-a122-cancel'
python = '/root/amadeus-m122l/w/bin/python'
sources = {n: source / n for n in ('pilot.py', 'private_capture.py', 'runtime-members.json')}
sources.update({n: base / n for n in ('auth_home.py', 'route.py')})
hashes = {n: hashlib.sha256(p.read_bytes()).hexdigest() for n, p in sources.items()}
assert hashes['pilot.py'] == '3910f6f6cac1e9bae824d44081e16cf4435400dbd109b94405d7acbb87179b3d'
assert hashes['private_capture.py'] == 'd9548ad05fab5c91ffb3f7f17ed86169881968af0d8b5a763f22af0ed18646d7'
c = paramiko.SSHClient()
c.load_host_keys('D:/agentsync-debug/vm/known_hosts')
c.connect('127.0.0.1', port=22222, username='root', key_filename='D:/agentsync-debug/vm/debug_ed25519', look_for_keys=False, allow_agent=False, timeout=15)
with c.open_sftp() as sftp:
    with sftp.open('/proc/sys/kernel/random/boot_id', 'rb') as f:
        boot = f.read().decode().strip()
    assert 'amadeus-lifecycle-a122-cancel' not in sftp.listdir('/root')
    assert 'amadeus-lifecycle-a122-timeout' not in sftp.listdir('/root')
    assert 'amadeus-lifecycle-a122-helpers' not in sftp.listdir('/root')
    sftp.mkdir(remote, mode=0o700)
    for name, path in sources.items():
        with sftp.open(remote + '/' + name, 'wx') as f:
            sftp.chmod(remote + '/' + name, 0o600)
            f.write(path.read_bytes())
        with sftp.open(remote + '/' + name, 'rb') as f:
            assert hashlib.sha256(f.read()).hexdigest() == hashes[name]
command = ' '.join(map(shlex.quote, (python, '-I', '-B', remote + '/pilot.py', 'prepare', 'cancel')))
with (base / 'prepare-attempt.json').open('x', encoding='utf8') as f:
    json.dump({'command': command, 'boot_id': boot, 'helper_sha256': hashes, 'model_calls': 0, 'credential_access': 0}, f, indent=2)
_, stdout, stderr = c.exec_command(command, timeout=45)
raw, error = stdout.read(), stderr.read()
code = stdout.channel.recv_exit_status()
for name, data in [('prepare.stdout.log', raw), ('prepare.stderr.log', error)]:
    with (base / name).open('xb') as f:
        f.write(data)
assert code == 0, {'exit_code': code, 'stdout': raw.decode(errors='replace'), 'stderr': error.decode(errors='replace')}
with c.open_sftp() as sftp:
    info = sftp.lstat(root + '/plan.json')
    assert stat.S_ISREG(info.st_mode) and stat.S_IMODE(info.st_mode) == 0o600 and info.st_uid == 0
    with sftp.open(root + '/plan.json', 'rb') as f:
        plan_raw = f.read(65537)
    assert len(plan_raw) <= 65536
    assert not sftp.listdir(root + '/private-home')
    assert not sftp.listdir(root + '/receipts')
c.close()
plan = json.loads(plan_raw)
assert plan['mode'] == 'cancel' and plan['timeout_seconds'] == 90 and plan['model_calls'] == 0
with (base / 'plan-cancel.json').open('xb') as f:
    f.write(plan_raw)
control = {'mode': 'cancel', 'plan_sha256': plan['plan_sha256'],
           'plan_file_sha256': hashlib.sha256(plan_raw).hexdigest(), 'boot_id': boot,
           'helper_sha256': hashes, 'max_model_calls': 1, 'max_candidate_executions': 0,
           'subscription_only': True, 'model_calls': 0, 'credential_copy': False}
with (base / 'control-cancel.json').open('x', encoding='utf8') as f:
    json.dump(control, f, indent=2)
with (base / 'prepare-result.json').open('x', encoding='utf8') as f:
    json.dump({'exit_code': code, 'result': json.loads(raw), 'control': control}, f, indent=2)
print(json.dumps({'prepared': True, 'mode': 'cancel', 'plan_sha256': control['plan_sha256'],
                  'plan_file_sha256': control['plan_file_sha256'], 'boot_id': boot,
                  'model_calls': 0, 'credential_access': 0}))
