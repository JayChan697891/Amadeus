def cases(path):
    outcomes = {}
    for node in ET.parse(path).iter("testcase"):
        case = (node.get("classname"), node.get("name"))
        assert all(case) and case not in outcomes
        outcomes[case] = next(
            (child.tag for child in node if child.tag in {"failure", "error", "skipped"}), "pass"
        )
    return outcomes


def main():
    if not __debug__ or DEST.exists() or SUMMARY.exists():
        raise SystemExit("Use fresh collection paths with assertions enabled.")
    client = paramiko.SSHClient()
    client.load_host_keys("D:/agentsync-debug/vm/known_hosts")
    client.set_missing_host_key_policy(paramiko.RejectPolicy())
    try:
        client.connect(
            "127.0.0.1", port=22222, username="root", timeout=15,
            key_filename="D:/agentsync-debug/vm/debug_ed25519", look_for_keys=False, allow_agent=False,
        )
        stdin, stdout, stderr = client.exec_command("/root/jobs-env/bin/python -I -B -", timeout=180)
        stdin.write(PROBE)
        stdin.flush()
        stdin.channel.shutdown_write()
        output, errors = stdout.read(), stderr.read()
        assert stdout.channel.recv_exit_status() == 0 and not errors, "Guest terminal check refused collection."
        record = json.loads(output)
        assert BASELINE.stat().st_size == 394535
        assert hashlib.sha256(BASELINE.read_bytes()).hexdigest() == BASELINE_SHA
        wheel = cases(BASELINE)
        assert Counter(wheel.values()) == {"pass": 2589, "skipped": 7}
        DEST.mkdir()
        with client.open_sftp() as sftp:
            for entry in record["files"]:
                name = PurePosixPath(entry["path"])
                assert not name.is_absolute() and ".." not in name.parts
                assert "\\" not in entry["path"] and ":" not in entry["path"]
                remote = f"{GUEST}/run/{name.as_posix()}"
                info = sftp.lstat(remote)
                assert stat.S_ISREG(info.st_mode) and info.st_size == entry["bytes"]
                path = DEST.joinpath(*name.parts)
                path.parent.mkdir(parents=True, exist_ok=True)
                digest = hashlib.sha256()
                size = 0
                with sftp.open(remote, "rb") as source, path.open("xb") as destination:
                    while chunk := source.read(1024 * 1024):
                        size += len(chunk)
                        digest.update(chunk)
                        destination.write(chunk)
                assert (size, digest.hexdigest()) == (entry["bytes"], entry["sha256"])
        sdist_junit = DEST / "sdist-junit.xml"
        if sdist_junit.is_file():
            sdist = cases(sdist_junit)
            record["linux_wheel_comparison"] = {
                "wheel_baseline_sha256": BASELINE_SHA,
                "wheel_cases": len(wheel), "sdist_cases": len(sdist),
                "sdist_outcome_counts": dict(Counter(sdist.values())),
                "case_ids_match": wheel.keys() == sdist.keys(),
                "matching_case_outcomes": sum(sdist.get(case) == value for case, value in wheel.items()),
                "missing_case_ids": sorted(wheel.keys() - sdist.keys()),
                "new_case_ids": sorted(sdist.keys() - wheel.keys()),
                "outcome_changes": [
                    {"case": case, "wheel": wheel[case], "sdist": sdist[case]}
                    for case in sorted(wheel.keys() & sdist.keys()) if wheel[case] != sdist[case]
                ],
            }
        record["collected_at"] = datetime.now(UTC).isoformat()
        with SUMMARY.open("x", encoding="utf-8", newline="\n") as stream:
            json.dump(record, stream, indent=2)
            stream.write("\n")
        print(json.dumps({
            "run_status": record["run_status"], "files": len(record["files"]),
            "bytes": sum(entry["bytes"] for entry in record["files"]), "summary": str(SUMMARY),
        }))
    finally:
        client.close()


if __name__ == "__main__":
    main()
