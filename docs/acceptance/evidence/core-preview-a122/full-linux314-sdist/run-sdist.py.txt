TEST_COMMIT = "5232e9e8bd33a949f7ffdc68410cd614a297af66"

PINNED = {

    MANIFEST: (762297, "5ed855d481379438b4e2a70576c8a8befe5ff1a193219b4e3fa0eb4c0b459c78"),
    MEMBERS: (10144, "cdc35d4b827a54bbe8e47e69e994216f1a1d401ae5ca522fc4f041162cf48e2c"),
    SDIST: (6735308, SDIST_SHA),
}
SOURCE_FILES = 3992
COLLECTOR_SHA = "d942383c3a90b174149a9df5638440e5661f4279149c9bfd004c785d2cb680dd"

EVIDENCE_GROUPS = {
    "VERIFIER": 14, "OWNER_EXECUTOR": 7, "RECONCILIATION": 5, "CANDIDATE_EXECUTOR": 5,

ORIGIN_PROBE = r"""
import hashlib, importlib.metadata as md, json, sys, sysconfig
from pathlib import Path
import agentsync, agentsync.cli

install, archive, members = map(Path, sys.argv[1:4])
archive_sha, version = sys.argv[4:]
assert sys.version_info[:3] == (3, 14, 7)
assert Path(sys.prefix).resolve() == install.resolve()
site = Path(sysconfig.get_path("purelib")).resolve()

direct = json.loads(dist.read_text("direct_url.json"))

        with path.open("rb") as stream:
            actual[path.relative_to(site).as_posix()] = hashlib.file_digest(stream, "sha256").hexdigest()

                  "runtime_members": len(actual), "direct_url": direct}))

def sha(path):
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def save(path, value):
    with path.open("x", encoding="utf-8", newline="\n") as stream:
        json.dump(value, stream, indent=2)
        stream.write("\n")


def verify_inputs():
    for path, expected in PINNED.items():
        assert path.resolve(strict=True) == path and stat.S_ISREG(path.lstat().st_mode), path.name
        assert (path.stat().st_size, sha(path)) == expected, path.name


def extract_source(manifest):
    SOURCE.mkdir()
    with zipfile.ZipFile(SNAPSHOT) as archive:
        assert archive.comment == TEST_COMMIT.encode("ascii")
        entries = [item for item in archive.infolist() if not item.is_dir()]
        assert len(entries) == SOURCE_FILES and {entry.filename for entry in entries} == manifest.keys()
        for entry in entries:
            name = entry.filename
            path = SOURCE / name
            assert path.resolve().is_relative_to(SOURCE) and not stat.S_ISLNK(entry.external_attr >> 16)
            assert entry.file_size == manifest[name]["bytes"], name
            path.parent.mkdir(parents=True, exist_ok=True)
            with archive.open(entry) as origin, path.open("xb") as destination:
                shutil.copyfileobj(origin, destination)


def verify_source(manifest):
    names = set()
    for path in SOURCE.rglob("*"):
        assert path.resolve(strict=True) == path
        info = path.lstat()
        if stat.S_ISDIR(info.st_mode):
            continue
        assert stat.S_ISREG(info.st_mode)
        name = path.relative_to(SOURCE).as_posix()
        assert manifest[name] == {"sha256": sha(path), "bytes": info.st_size}, name
        names.add(name)
    assert names == manifest.keys() and len(names) == SOURCE_FILES


def isolate_environment():
    for name in tuple(os.environ):
        upper = name.upper()
        if upper.startswith((
            "AGENTSYNC_", "ANTHROPIC_", "CLAUDE_", "CODEX_", "CURSOR_", "DEEPSEEK_", "DOCKER_",
            "GEMINI_", "GIT_", "GITHUB_", "GOOGLE_", "OPENAI_", "PIP_", "PYTEST_", "PYTHON", "UV_", "XDG_",
        )) or upper.endswith(("API_KEY", "ACCESS_TOKEN", "REFRESH_TOKEN")) or upper == "SSH_AUTH_SOCK":
            os.environ.pop(name)
    os.environ.update(
        HOME=str(HOME), USERPROFILE=str(HOME), CODEX_HOME=str(HOME),
        DOCKER_CONFIG=str(HOME / ".docker"), DOCKER_HOST="unix:///var/run/docker.sock",
        XDG_CONFIG_HOME=str(HOME / "config"), XDG_CACHE_HOME=str(HOME / "cache"),
        TEMP=str(SCRATCH), TMP=str(SCRATCH), TMPDIR=str(SCRATCH),
        PYTHONDONTWRITEBYTECODE="1", PYTHONNOUSERSITE="1",
        GIT_CONFIG_NOSYSTEM="1", GIT_CONFIG_GLOBAL=os.devnull, GIT_TERMINAL_PROMPT="0",
        GCM_INTERACTIVE="Never", PIP_CONFIG_FILE=os.devnull, PIP_NO_INDEX="1",
        PIP_DISABLE_PIP_VERSION_CHECK="1", PIP_NO_CACHE_DIR="1",
        RUFF_CACHE_DIR=str(SCRATCH / "ruff"), MYPY_CACHE_DIR=str(SCRATCH / "mypy"),
    )


def docker(*arguments):
    for executable in ("/usr/bin/docker", "/usr/local/bin/docker"):
        if os.access(executable, os.X_OK):
            return subprocess.check_output([executable, *arguments], text=True, timeout=30)
    raise FileNotFoundError("Local Docker CLI is unavailable")


def docker_state():
    return {
        "containers": docker("ps", "-aq", "--no-trunc").splitlines(),
        "image_id": docker("image", "inspect", "--format", "{{.Id}}", DOCKER_IMAGE).strip(),
    }


def docker_evidence(root, redact):
    paths = sorted(root.rglob("*.json"))
    groups = {name: 0 for name in EVIDENCE_GROUPS}
    assert len(paths) == sum(EVIDENCE_GROUPS.values()) and sum(path.stat().st_size for path in paths) <= 4 * 1024 * 1024
    hashes = {}
    for path in paths:
        group = path.relative_to(root).parts[0].upper()
        groups[group] += 1
        assert path.resolve(strict=True) == path and redact(path.read_text(encoding="utf-8"))[1] == 0
        hashes[path.relative_to(RUN).as_posix()] = sha(path)
    assert groups == EVIDENCE_GROUPS

    verifier, = (root / "verifier").glob("*/test_actual_docker_checks_all_external_cases.json")
    direct = json.loads(verifier.read_bytes())
    assert direct["image_id"] == DOCKER_IMAGE and direct["cleanup_confirmed"] is True
    for name, verdict in (("verified", "verified"), ("wrong-result", "unverified")):
        pilot = json.loads((root / "proposal_pipeline" / f"{name}.json").read_bytes())
        execution = pilot["execution"]["execution"]["result"]
        assert pilot["model_calls"] == 0 and pilot["task_verification"] == verdict
        assert pilot["container_absent"] is True and execution["cleanup_confirmed"] is True
        assert execution["backend"] == "local-linux-docker" and execution["image_sha256"] == DOCKER_IMAGE
    continuation = json.loads((root / "proposal_pipeline/continuation-verified.json").read_bytes())
    execution = continuation["execution"]["execution"]["result"]
    assert continuation["model_calls"] == 0 and continuation["same_core"] is True
    assert continuation["source_task_status"] == "failed" and continuation["receiver_task_status"] == "completed"
    assert continuation["receiver_task_verification"] == "verified"
    assert continuation["completion"]["task_id"] == continuation["receiver_task_id"]
    assert execution["backend"] == "local-linux-docker" and execution["image_sha256"] == DOCKER_IMAGE
    assert execution["cleanup_confirmed"] is True
    return hashes


def main():
    if (
        sys.platform != "linux" or sys.version_info[:3] != (3, 14, 7)
        or Path(sys.executable) != PYTHON or Path(sys.prefix) != INSTALL
        or not sys.flags.isolated or not sys.dont_write_bytecode or not __debug__
    ):
        raise SystemExit(f"Use the installed a122 Linux 3.14.7 interpreter with -I -B: {PYTHON}")
    if len(sys.argv) != 2 or len(sys.argv[1]) != 64 or any(c not in "0123456789abcdef" for c in sys.argv[1]):
        raise SystemExit("Supply the SHA-256 of the newly reviewed guest-preflight.json.")
    preflight_sha = sys.argv[1]
    script = Path(__file__).resolve(strict=True)
    assert BASE.resolve(strict=True) == BASE and script == BASE / "run-sdist.py"
    expected_files = {path.name for path in (script, PREFLIGHT)}
    if {path.name for path in BASE.iterdir()} != expected_files:
        raise SystemExit("A fresh driver directory is required. Preserve any existing attempt.")
    os.umask(0o077)
    RUN.mkdir()
    start = {
        "scope": "Linux 3.14 full installed a122 sdist suite with actual local Docker",
        "test_source_commit": TEST_COMMIT, "product_runtime_commit": PRODUCT_COMMIT, "version": VERSION,
        "pinned_inputs": {str(path): {"bytes": size, "sha256": digest} for path, (size, digest) in PINNED.items()},
        "driver_sha256": sha(script), "controller_pid": os.getpid(), "python": sys.version,
        "python_executable": sys.executable, "boot_id": Path("/proc/sys/kernel/random/boot_id").read_text().strip(),
        "required_docker_image_id": DOCKER_IMAGE, "model_calls": 0,
        "guest_preflight_sha256": preflight_sha,
        "started_at": datetime.now(UTC).isoformat(), "status": "unknown", "automatic_retry": False,
        "formal_release": "BLOCKED", "real_clients": "NOT RUN", "full_platform_matrix": "NOT RUN",
    }
    save(RUN / "attempt.json", start)
    result = {**start, "status": "fail", "phase": "prepare", "checks": []}
    source_verified = docker_inspected = False
    empty_daemon = {"image_id": DOCKER_IMAGE, "containers": []}
    try:
        (RUN / "logs").mkdir()
        SCRATCH.mkdir()
        HOME.mkdir()
        isolate_environment()
        assert PREFLIGHT.resolve(strict=True) == PREFLIGHT and stat.S_ISREG(PREFLIGHT.lstat().st_mode)
        assert sha(PREFLIGHT) == preflight_sha
        preflight = json.loads(PREFLIGHT.read_bytes())
        assert preflight["boot_id"] == start["boot_id"]
        assert preflight["workers"] == preflight["containers"] == []
        assert preflight["runtime_members"] == 97 and preflight["image_id"] == DOCKER_IMAGE
        assert preflight["inputs"] == {path.name: digest for path, (_, digest) in PINNED.items()}
        direct = preflight["direct_url"]
        assert direct["url"] == SDIST.as_uri() and "dir_info" not in direct
        assert direct["archive_info"]["hashes"]["sha256"] == SDIST_SHA
        verify_inputs()
        manifest = json.loads(MANIFEST.read_bytes())
        assert len(manifest) == SOURCE_FILES
        extract_source(manifest)
        verify_source(manifest)
        source_verified = True
        members = json.loads(MEMBERS.read_bytes())
        assert len(members) == 97 and members == {
            name.removeprefix("src/"): row["sha256"]
            for name, row in manifest.items() if name.startswith("src/agentsync/")
        }
        assert manifest["tools/release_acceptance.py"]["sha256"] == COLLECTOR_SHA
        result.update(source_files=len(manifest), runtime_members=len(members), collector_sha256=COLLECTOR_SHA)
        docker_inspected = True
        result["docker_before"] = docker_state()
        assert result["docker_before"] == empty_daemon
        sys.path.insert(0, str(SOURCE))
        from tools import release_acceptance as collector

        assert Path(collector.__file__).resolve() == SOURCE / "tools/release_acceptance.py"
        def check(name, command, **options):
            result["phase"] = name
            observed = collector.execute(name, list(map(str, command)), RUN, RUN, SCRATCH, **options)
            result["checks"].append(observed)
            if observed["exit_code"] is None:
                result["status"] = "unknown"
            return observed

        def require(observed):
            assert observed["status"] == "pass" and observed["exit_code"] == 0
            assert observed["log_truncated"] is False

        python = [sys.executable, "-I", "-B"]
        origin = [*python, "-c", ORIGIN_PROBE, INSTALL, SDIST, MEMBERS, SDIST_SHA, VERSION]
        require(check("sdist_origin_before", origin))

        observations = RUN / "observations"
        for group in EVIDENCE_GROUPS:
            path = observations / group.lower()
            path.mkdir(parents=True)
            os.environ[f"AGENTSYNC_{group}_EVIDENCE_DIR"] = str(path)
        os.environ.update(AGENTSYNC_DOCKER_TEST_IMAGE=DOCKER_IMAGE, AGENTSYNC_REQUIRE_DOCKER_TESTS="1")
        raw = SCRATCH / "sdist-junit.xml"
        junit = RUN / raw.name
        stdout = SCRATCH / "sdist-pytest-stdout.raw.log"
        pytest_command = [
            *python, "-m", "pytest", str(SOURCE / "tests"), "--junitxml", str(raw),
            "--basetemp", str(SCRATCH / "p"), "-o", f"cache_dir={SCRATCH / 'pytest-cache'}",
        ]
        save(RUN / "pytest-start.json", {
            "controller_pid": os.getpid(), "boot_id": start["boot_id"], "argv": pytest_command,
            "requested_at": datetime.now(UTC).isoformat(), "timeout_seconds": 7200, "cwd": str(RUN),
            "test_source_commit": TEST_COMMIT, "pinned_inputs": start["pinned_inputs"],
            "required_docker_image_id": DOCKER_IMAGE,
        })
        try:
            tested = check("sdist_pytest", pytest_command, pytest_timeout=7200, capture_stdout=stdout)
        finally:
            try:
                if raw.is_file():
                    result["raw_junit_local"] = {"bytes": raw.stat().st_size, "sha256": sha(raw)}
                    redactions = collector.persist_junit(raw, junit)
                    counts = collector.junit_counts(junit)
                    cases = [(node.get("classname"), node.get("name")) for node in ET.parse(junit).iter("testcase")]
                    assert len(cases) == counts["tests"] == len(set(cases)) and all(all(case) for case in cases)
                    result["junit"] = {"path": junit.name, "sha256": sha(junit),
                                       "credential_redactions": redactions, **counts}
            except Exception as exc:
                result["junit_error_type"] = type(exc).__name__
            try:
                if stdout.is_file():
                    result["raw_stdout_local"] = {"bytes": stdout.stat().st_size, "sha256": sha(stdout)}
                    with stdout.open("rb") as stream:
                        output = stream.read(collector.MAX_LOG_BYTES + 1)
                    truncated = len(output) > collector.MAX_LOG_BYTES
                    clean, redactions = collector.redact_credentials(
                        output[:collector.MAX_LOG_BYTES].decode("utf-8", errors="replace")
                    )
                    log = RUN / "logs/sdist_pytest.stdout.log"
                    with log.open("x", encoding="utf-8") as stream:
                        stream.write(clean)
                        if truncated:
                            stream.write("\n[OUTPUT TRUNCATED: evidence processing limit reached]\n")
                    result["pytest_stdout"] = {
                        "path": log.relative_to(RUN).as_posix(), "sha256": sha(log),
                        "credential_redactions": redactions, "truncated": truncated,
                    }
            except Exception as exc:
                result["stdout_error_type"] = type(exc).__name__
            try:
                origin_after = check("sdist_origin_after", origin)
            except Exception as exc:
                result["origin_after_error_type"] = type(exc).__name__
                origin_after = None
        result["phase"] = "full_suite_evidence"
        require(tested)
        assert origin_after is not None
        require(origin_after)
        before, after = (RUN / "logs" / f"sdist_origin_{phase}.log" for phase in ("before", "after"))
        assert before.read_bytes() == after.read_bytes()
        assert result["pytest_stdout"]["truncated"] is False
        counts = result["junit"]
        assert counts["tests"] > 0 and counts["failures"] == counts["errors"] == 0
        result["fixture_observations_sha256"] = docker_evidence(observations, collector.redact_credentials)
        result["status"] = "pass"
    except KeyboardInterrupt:
        result.update(status="unknown", error_type="KeyboardInterrupt")
    except Exception as exc:
        result["error_type"] = type(exc).__name__
    finally:
        if docker_inspected:
            try:
                result["docker_after"] = docker_state()
                assert result["docker_after"] == empty_daemon
            except Exception as exc:
                if result["status"] == "pass":
                    result["status"] = "fail"
                result["docker_after_error_type"] = type(exc).__name__
        try:
            verify_inputs()
            result["pinned_artifacts_unchanged"] = True
            if source_verified:
                verify_source(manifest)
                result["frozen_source_unchanged"] = True
        except Exception as exc:
            if result["status"] == "pass":
                result["status"] = "fail"
            result["postcheck_error_type"] = type(exc).__name__
        result["finished_at"] = datetime.now(UTC).isoformat()
        save(RUN / "result.json", result)
        print(f"{result['status']}: {RUN / 'result.json'}", flush=True)
    return 0 if result["status"] == "pass" else 1


if __name__ == "__main__":
    raise SystemExit(main())
