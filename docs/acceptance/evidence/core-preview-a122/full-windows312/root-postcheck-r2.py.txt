def cases(path):
    data = {}
    skips = collections.Counter()
    for node in ET.parse(path).iter('testcase'):
        key = node.attrib['classname'] + '::' + node.attrib['name']
        assert key not in data
        outcome = next((kind for kind in ['failure','error','skipped'] if node.find(kind) is not None),'pass')
        data[key] = outcome
        if outcome == 'skipped':
            skips[node.find('skipped').attrib.get('message','')] += 1
    return data, skips

current, skips = cases(run/'wheel-junit.xml')
previous_path = repo/'docs/acceptance/evidence/core-preview-a120/full-windows312/wheel-junit.xml'
previous, _ = cases(previous_path)
