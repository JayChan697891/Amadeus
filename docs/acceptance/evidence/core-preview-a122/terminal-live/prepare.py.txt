"""Prepare a new private fixture; never read account auth or run a model."""
import hashlib
import json
import os
import subprocess
import sys
from datetime import UTC, datetime
from pathlib import Path

import agentsync
from agentsync.core.service import Core

BASE = Path('/root/amadeus-terminal-live-a122-20261118')
EXE = Path('/root/amadeus-codex-linux-0167/bin/codex')
BINARY = 'c8242fc9e1bac7c6d661a228cecfaf50e6db74da219e4a9ca8912827e11607f0'
IMAGE = 'sha256:34386ef0cb081344d7ec1c103ba398e6e9f64e9ab3a1509accc92a4e24a07258'
EXAMPLE = 'c4540ffda772877e4516b12b7673e256b70d4f9f90c6b39dc5638efcc5673ae9'
SOURCE = '8172bd124ebb1bb6a3b71170a7e40e0beab67145a24801e14e9c7887a55c8b44'

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def save(name, value):
    with (BASE / name).open('x', encoding='utf8') as stream:
        json.dump(value, stream, indent=2)
        stream.flush()
        os.fsync(stream.fileno())

def main():
    os.umask(0o077)
    assert BASE.resolve(strict=True) == BASE
    assert sha(BASE/'verified_codex_task.py') == EXAMPLE
    assert sha(EXE) == BINARY
    expected = json.loads((BASE/'runtime-members.json').read_text())
    site = Path(agentsync.__file__).resolve().parents[1]
    assert len(expected) == 97
    assert {name:sha(site/name) for name in expected} == expected
    candidate = json.loads((BASE/'candidate-source.json').read_text())
    assert candidate['path'] == 'solution.py'
    assert len(candidate['content'].encode()) == 149
    assert hashlib.sha256(candidate['content'].encode()).hexdigest() == SOURCE
    for name in ('source','private-home','preflight-home'):
        (BASE/name).mkdir(mode=0o700)
    (BASE/'source/baseline.txt').write_text('Dedicated terminal acceptance project.\n')
    clean = {'PATH':'/usr/bin:/bin','HOME':str(BASE/'preflight-home'),
             'CODEX_HOME':str(BASE/'preflight-home'),'LANG':'C.UTF-8'}
    commands = [
        ['/usr/bin/git','init','--initial-branch=terminal-fixture'],
        ['/usr/bin/git','add','baseline.txt'],
        ['/usr/bin/git','-c','user.name=Amadeus fixture','-c','user.email=fixture@example.invalid',
         'commit','-m','Dedicated terminal fixture'],
    ]
    for argv in commands:
        subprocess.run(argv,cwd=BASE/'source',env=clean,check=True,capture_output=True,timeout=20)
    version = subprocess.run([str(EXE),'--version'],cwd=BASE,env=clean,check=True,
                             capture_output=True,text=True,timeout=20).stdout.strip()
    assert version == 'codex-cli 0.167.0-alpha.3.2'
    inspected = subprocess.run(['/usr/bin/docker','image','inspect',IMAGE],env=clean,
                               check=True,capture_output=True,text=True,timeout=20)
    image, = json.loads(inspected.stdout)
    assert image['Id'] == IMAGE
    assert not (BASE/'private-home/auth.json').exists()
    assert not (BASE/'terminal-run').exists()
    owner = Core(BASE/'core').execute('create_project',{'name':'Terminal exact-source trial',
                                                      'root_path':str(BASE/'source')})
    save('private-owner.json',owner)
    task = ('Return solution.py with exactly these UTF-8 source bytes, including the final newline. '
            'Do not improve, reformat or execute it.\n\n' + candidate['content'])
    assert len(task.encode()) <= 1000
    with (BASE/'task.txt').open('x',encoding='utf8') as f:f.write(task)
    cases = [{'input':[2,3],'expected':5},{'input':[-4,1],'expected':-3}]
    save('cases.json',cases)
    argv = [sys.executable,'-I','-B',str(BASE/'verified_codex_task.py'),
            '--root',str(BASE),'--project-id',owner['project_id'],'--task',str(BASE/'task.txt'),
            '--cases',str(BASE/'cases.json'),'--codex',str(EXE),'--codex-sha256',BINARY,
            '--codex-version','0.167.0-alpha.3.2','--codex-home',str(BASE/'private-home'),
            '--image-id',IMAGE]
    ready = {'created_at':datetime.now(UTC).isoformat(),'project_id':owner['project_id'],
             'fixture':str(BASE),'command':argv,'example_commit':'13eb0759138de485e2ebba445e93369a28e3c916',
             'example_sha256':EXAMPLE,'runtime_commit':'5232e9e8bd33a949f7ffdc68410cd614a297af66',
             'installed_members':97,'binary_sha256':BINARY,'cli_version':version,
             'image_id':IMAGE,'candidate_sha256':SOURCE,'candidate_bytes':149,'cases':cases,
             'max_model_calls':2,'max_candidate_executions':1,'retry_allowed':False,
             'files_sha256':{name:sha(BASE/name) for name in ('verified_codex_task.py','task.txt',
                       'cases.json','candidate-source.json','runtime-members.json','prepare.py')},
             'model_calls':0,'candidate_executions':0,'auth_copied':False,
             'candidate_execution_authorized':False,'formal_release':'BLOCKED'}
    save('ready.json',ready)
    print(json.dumps(ready))

if __name__ == '__main__':
    main()
