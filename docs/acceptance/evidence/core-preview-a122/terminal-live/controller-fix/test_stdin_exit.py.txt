"""Local stdin transport regression only; no SSH, PTY or model execution."""

import ast
import os
import queue
import subprocess
import sys
import threading
from pathlib import Path

FIXED = Path("D:/agentsync-debug/terminal-live-a122-20261118/terminal_control_fixed.py")
NODE = next(node for node in ast.parse(FIXED.read_bytes()).body
            if isinstance(node, ast.FunctionDef) and node.name == "read_manual")
NAMESPACE = {"os": os}
exec(compile(ast.Module(body=[NODE], type_ignores=[]), str(FIXED), "exec"), NAMESPACE)
READ = NAMESPACE["read_manual"]


def test_remote_terminal_exit_with_local_stdin_open():
    code = (
        "import ast,os,queue,sys,threading,time; from pathlib import Path; "
        f"path=Path({str(FIXED)!r}); "
        "node=next(n for n in ast.parse(path.read_bytes()).body "
        "if isinstance(n,ast.FunctionDef) and n.name=='read_manual'); "
        "ns={'os':os}; exec(compile(ast.Module(body=[node],type_ignores=[]),str(path),'exec'),ns); "
        "read=ns['read_manual']; "
        "q=queue.Queue(); "
        "threading.Thread(target=read,args=(q,sys.stdin.fileno()),daemon=True).start(); "
        "time.sleep(0.1); print('remote terminal observed; local stdin stays open',flush=True)"
    )
    process = subprocess.Popen(
        [sys.executable, "-I", "-B", "-c", code], stdin=subprocess.PIPE,
        stdout=subprocess.PIPE, stderr=subprocess.PIPE,
        creationflags=getattr(subprocess, "CREATE_NO_WINDOW", 0),
    )
    try:
        assert process.stdout.readline() == b"remote terminal observed; local stdin stays open\r\n"
        assert not process.stdin.closed
        assert process.wait(timeout=10) == 0
        assert not process.stdin.closed
        assert process.stderr.read() == b""
    finally:
        if process.poll() is None:
            process.kill()
            process.wait(timeout=5)
        process.stdin.close()
        process.stdout.close()
        process.stderr.close()


def test_normal_lines_forwarded_without_changes():
    reader, writer = os.pipe()
    pending = queue.Queue()
    thread = threading.Thread(target=READ, args=(pending, reader))
    thread.start()
    try:
        os.write(writer, b"CONFIRM\n" + b"a" * 64 + b"\n")
        assert pending.get(timeout=5) == b"CONFIRM\n"
        assert pending.get(timeout=5) == b"a" * 64 + b"\n"
    finally:
        os.close(writer)
        thread.join(timeout=5)
        os.close(reader)
    assert not thread.is_alive() and pending.get(timeout=1) is None


def test_local_eof_notifies_once():
    reader, writer = os.pipe()
    pending = queue.Queue()
    os.close(writer)
    try:
        READ(pending, reader)
        assert pending.get_nowait() is None and pending.empty()
    finally:
        os.close(reader)
