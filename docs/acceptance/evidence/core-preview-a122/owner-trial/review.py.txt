import hashlib,json,sqlite3
from pathlib import Path
base=Path('D:/agentsync-debug/owner-trial-a122-20261117')
response=json.loads((base/'demo.stdout.json').read_text(encoding='utf-8-sig'))
assert response['ok'] is True and (base/'demo.stderr.log').stat().st_size==0
data=response['data']
assert data['simulation_only'] is True and data['verification']=='unverified'
assert data['resumed_in_fresh_process'] is True and data['attempts']==['failed','succeeded','succeeded']
connection=sqlite3.connect((base/'data/agentsync.db').as_uri()+'?mode=ro',uri=True)
try:
 task=connection.execute('SELECT id, status FROM tasks WHERE id=?',(data['task_id'],)).fetchone()
 workflow=connection.execute('SELECT id, state FROM mock_workflows WHERE id=?',(data['run_id'],)).fetchone()
 handoff=connection.execute('SELECT id,status,sender_agent_id,recipient_agent_id FROM handoffs WHERE id=?',(data['handoff']['id'],)).fetchone()
 reviews=connection.execute('SELECT reviewer_agent_id,accept FROM reviews WHERE task_id=?',(data['task_id'],)).fetchall()
 attempts=connection.execute('SELECT state FROM mock_workflow_attempts WHERE run_id=? ORDER BY sequence',(data['run_id'],)).fetchall()
finally:connection.close()
assert task==(data['task_id'],'completed') and workflow==(data['run_id'],'completed')
assert handoff[1]=='accepted' and len(reviews)==1 and reviews[0][1]==1
assert reviews[0][0] not in handoff[2:]
