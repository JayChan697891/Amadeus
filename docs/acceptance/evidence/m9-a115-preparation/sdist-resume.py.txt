"""Run only the previously unrun sdist checks on the existing frozen archive."""
from datetime import datetime, timezone
import hashlib
import json
import os
from pathlib import Path
import shutil
import subprocess
import sys
import tomllib

SOURCE = Path('D:/agentsync-debug/m115-r3/s')
ORIGINAL = Path('D:/agentsync-debug/m115-r3/windows-312/release-20261114T035240Z-6756b7e7fa0d')
CHECKPOINT = Path('F:/syuc/checkpoints/amadeus-safe-pause-20261114T045950Z')
BASE = Path('D:/agentsync-debug/m115-sdist-r1')
SCRATCH = Path('D:/agentsync-debug/t115v')
WHEELS = Path('D:/agentsync-debug/continuation-cc30ed21/windows-py312-wheelhouse')
COMMIT = 'b6f05f0473591dc568be25591047fdea963ef86c'
ARCHIVE_SHA = '407ab42221a5b07d8638d91382c3ec49cac139bb1c21e3c1eea0b0ddce061f56'

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()

def read(path):
    return json.loads(path.read_text(encoding='utf-8'))

def save(path, value):
    with path.open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')

assert sys.version_info[:2] == (3, 12)
assert not BASE.exists() and not SCRATCH.exists()
for key in tuple(os.environ):
    if key.upper().startswith(('OPENAI_', 'ANTHROPIC_', 'CLAUDE_', 'CURSOR_', 'CODEX_', 'AGENTSYNC_', 'GIT_', 'PYTEST_', 'PYTHON', 'PIP_', 'GOOGLE_', 'GEMINI_', 'DEEPSEEK_')) or key.upper().endswith(('API_KEY', 'ACCESS_TOKEN', 'REFRESH_TOKEN')):
        os.environ.pop(key, None)
os.environ.update(GIT_CONFIG_NOSYSTEM='1', GIT_CONFIG_GLOBAL=os.devnull, GIT_TERMINAL_PROMPT='0', PYTHONDONTWRITEBYTECODE='1', PYTHONNOUSERSITE='1', PYTHONUNBUFFERED='1', PIP_NO_INDEX='1', PIP_CONFIG_FILE=os.devnull, PIP_DISABLE_PIP_VERSION_CHECK='1')
assert subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=SOURCE, text=True).strip() == COMMIT
assert subprocess.check_output(['git', 'rev-parse', 'HEAD^{tree}'], cwd=SOURCE, text=True).strip() == '8a65e7d29ef72b877cca319beb45c243a0df05e1'
assert subprocess.check_output(['git', 'status', '--porcelain'], cwd=SOURCE) == b''
snapshot = read(ORIGINAL / 'source-snapshot.json')
for entry in snapshot:
    path = SOURCE / entry['path']
    assert path.stat().st_size == entry['size'] and sha(path) == entry['sha256'], entry['path']
build_receipt = ORIGINAL / 'logs/build.process.json'
build = read(build_receipt)
assert build['phase'] == 'finished' and build['result']['status'] == 'pass' and build['result']['exit_code'] == 0
assert build['result']['cwd'] == str(SOURCE)
assert sha(ORIGINAL / build['result']['log']) == build['result']['log_sha256']
archive = ORIGINAL / 'artifacts/agentsync_local-0.3.0a115.tar.gz'
assert sha(archive) == ARCHIVE_SHA and archive.stat().st_size == 2926414
checkpoint_pin = next(row for row in read(CHECKPOINT / 'progress-and-evidence-pins.json')['files'] if row['path'] == str(archive))
assert checkpoint_pin['sha256'] == ARCHIVE_SHA
assert sha(SOURCE / 'tools/release_acceptance.py') == 'd942383c3a90b174149a9df5638440e5661f4279149c9bfd004c785d2cb680dd'
sys.dont_write_bytecode = True
sys.path.insert(0, str(SOURCE))
from tools import release_acceptance as collector

BASE.mkdir()
SCRATCH.mkdir()
for name in ('home', 'tmp', 'run/logs', 'run/artifacts'):
    (BASE / name).mkdir(parents=True)
os.environ.update(HOME=str(BASE / 'home'), USERPROFILE=str(BASE / 'home'), CODEX_HOME=str(BASE / 'home'), TEMP=str(BASE / 'tmp'), TMP=str(BASE / 'tmp'), TMPDIR=str(BASE / 'tmp'))
run = BASE / 'run'
copied = run / 'artifacts' / archive.name
shutil.copyfile(archive, copied)
assert sha(copied) == ARCHIVE_SHA
save(BASE / 'attempt.json', {'pid': os.getpid(), 'started_at': datetime.now(timezone.utc).isoformat(), 'python': sys.version, 'executable': sys.executable, 'source_commit': COMMIT, 'source_tree': '8a65e7d29ef72b877cca319beb45c243a0df05e1', 'scope': 'Only previously unrun installed-sdist checks; no source/wheel/build rerun', 'archive_sha256': ARCHIVE_SHA, 'archive_size': copied.stat().st_size, 'original_build_receipt': str(build_receipt), 'original_build_receipt_sha256': sha(build_receipt), 'original_build_log_sha256': build['result']['log_sha256'], 'source_snapshot_sha256': sha(ORIGINAL / 'source-snapshot.json'), 'source_files_verified': len(snapshot), 'driver_sha256': sha(Path(__file__)), 'collector_sha256': sha(SOURCE / 'tools/release_acceptance.py'), 'automatic_retry': False, 'formal_release': 'BLOCKED'})
try:
    project = tomllib.loads((SOURCE / 'pyproject.toml').read_text(encoding='utf-8'))
    offline = collector._prepare_offline_packages(project, WHEELS, SCRATCH, run)
    assert read(run / 'offline-wheelhouse.json') == read(ORIGINAL / 'offline-wheelhouse.json')
    checks = collector.sdist_checks(SOURCE, run, SCRATCH, project['project']['version'], True, pytest_timeout=3600, offline_packages=offline, license_inventory=True)
    save(BASE / 'checks.json', checks)
    unchanged = all((SOURCE / item['path']).stat().st_size == item['size'] and sha(SOURCE / item['path']) == item['sha256'] for item in snapshot)
    passed = all(item['status'] == 'pass' for item in checks) and unchanged and sha(copied) == ARCHIVE_SHA
    result = {'finished_at': datetime.now(timezone.utc).isoformat(), 'exit_code': 0 if passed else 1, 'source_commit': COMMIT, 'archive_sha256': ARCHIVE_SHA, 'source_unchanged': unchanged, 'checks': [{'id': c['id'], 'status': c['status']} for c in checks], 'scope': 'Sdist component execution only; identity and composite evidence validation remain separate', 'formal_release': 'BLOCKED'}
    save(BASE / 'result.json', result)
    print(json.dumps(result), flush=True)
except BaseException as exc:
    save(BASE / 'exception.json', {'at': datetime.now(timezone.utc).isoformat(), 'error_type': type(exc).__name__, 'automatic_retry': False})
    raise
raise SystemExit(result['exit_code'])
