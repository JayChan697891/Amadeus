def read(path):
    return json.loads(path.read_text(encoding='utf-8'))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def write(name, data):
    path = TARGET.joinpath(*_member_parts(name))
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('xb') as stream:
        stream.write(data)
    return {'path': name, 'size': len(data), 'sha256': sha(path)}


def save(name, data):
    return write(name, (json.dumps(data, indent=2, allow_nan=False)+'\n').encode())
