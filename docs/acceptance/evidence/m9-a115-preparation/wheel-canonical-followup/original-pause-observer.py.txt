"""Owner-requested pause: observe an existing wheel process, never launch tests."""
import ctypes
from ctypes import wintypes
from datetime import datetime, timezone
import hashlib
import importlib.util
import json
import msvcrt
import os
from pathlib import Path
import subprocess
import time
import xml.etree.ElementTree as ET

OUT = Path('F:/syuc/checkpoints/amadeus-safe-pause-20261114T045950Z')
SCRATCH = Path('D:/agentsync-debug/t115u/.scratch-u_ak32ml')
SOURCE = Path('D:/agentsync-debug/m115-r3/s')
RUN = Path('D:/agentsync-debug/m115-r3/windows-312/release-20261114T035240Z-6756b7e7fa0d')
kernel = ctypes.WinDLL('kernel32', use_last_error=True)
kernel.OpenProcess.argtypes = [wintypes.DWORD, wintypes.BOOL, wintypes.DWORD]
kernel.OpenProcess.restype = wintypes.HANDLE
kernel.WaitForSingleObject.argtypes = [wintypes.HANDLE, wintypes.DWORD]
kernel.WaitForSingleObject.restype = wintypes.DWORD
kernel.GetExitCodeProcess.argtypes = [wintypes.HANDLE, ctypes.POINTER(wintypes.DWORD)]
kernel.GetProcessTimes.argtypes = [wintypes.HANDLE] + [ctypes.POINTER(wintypes.FILETIME)] * 4
kernel.CreateFileW.argtypes = [wintypes.LPCWSTR, wintypes.DWORD, wintypes.DWORD, wintypes.LPVOID, wintypes.DWORD, wintypes.DWORD, wintypes.HANDLE]
kernel.CreateFileW.restype = wintypes.HANDLE
kernel.CloseHandle.argtypes = [wintypes.HANDLE]

def save(name, data):
    with (OUT / name).open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(data, stream, indent=2)
        stream.write('\n')

command = "Get-CimInstance Win32_Process -Filter 'ProcessId = 14416' | Select-Object ProcessId,@{Name='CreationUtc';Expression={$_.CreationDate.ToUniversalTime().ToString('o')}},CommandLine | ConvertTo-Json -Compress"
identity = json.loads(subprocess.check_output(['powershell.exe', '-NoProfile', '-NonInteractive', '-Command', command], text=True, timeout=15))
assert identity['ProcessId'] == 14416 and identity['CreationUtc'] == '2026-11-14T04:35:22.1782290Z'
assert 'wheel-junit.xml' in identity['CommandLine'] and 't115u' in identity['CommandLine'] and '-m pytest' in identity['CommandLine']
handle = kernel.OpenProcess(0x101000, False, 14416)
assert handle
save('wheel-boundary-observer.json', {'pid': os.getpid(), 'started_at': datetime.now(timezone.utc).isoformat(), 'observed_process': identity, 'role': 'Wait for existing test to finish; no new task dispatch'})
try:
    deadline = time.monotonic() + 5400
    while kernel.WaitForSingleObject(handle, 1000) == 258:
        if time.monotonic() >= deadline:
            raise TimeoutError('Pause observation timed out; process not terminated')
    code = wintypes.DWORD()
    assert kernel.GetExitCodeProcess(handle, ctypes.byref(code))
    times = [wintypes.FILETIME() for _ in range(4)]
    assert kernel.GetProcessTimes(handle, *[ctypes.byref(value) for value in times])
    exited = ((times[1].dwHighDateTime << 32) | times[1].dwLowDateTime) / 10_000_000 - 11_644_473_600
    result = {'scope': 'Independent safe-pause observation; not a completed canonical collector receipt', 'exit_code': code.value, 'exited_at': datetime.fromtimestamp(exited, timezone.utc).isoformat(), 'observed_at': datetime.now(timezone.utc).isoformat(), 'pytest_pid': 14416, 'collector': 'Suspended to prevent sdist dispatch; original receipt stays started/unknown', 'artifacts': {}, 'formal_release': 'BLOCKED'}
    spec = importlib.util.spec_from_file_location('collector', SOURCE / 'tools/release_acceptance.py')
    collector = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(collector)
    raw = SCRATCH / 'wheel-junit.xml'
    if raw.is_file():
        result['raw_junit_sha256'] = hashlib.sha256(raw.read_bytes()).hexdigest()
        result['raw_junit_path_local_only'] = str(raw)
        destination = OUT / 'wheel-junit-sanitized.xml'
        collector.persist_junit(raw, destination)
        cases = list(ET.parse(destination).getroot().iter('testcase'))
        raw_cases = list(ET.parse(raw).getroot().iter('testcase'))
        result['junit'] = {'records': len(cases), 'raw_unique': len({(c.get('classname'), c.get('name')) for c in raw_cases}), 'sanitized_unique': len({(c.get('classname'), c.get('name')) for c in cases}), 'failed': sum(c.find('failure') is not None for c in cases), 'errors': sum(c.find('error') is not None for c in cases), 'skipped': sum(c.find('skipped') is not None for c in cases)}
        result['junit']['passed'] = len(cases) - sum(result['junit'][field] for field in ('failed', 'errors', 'skipped'))
        result['artifacts'][destination.name] = hashlib.sha256(destination.read_bytes()).hexdigest()
    log_handle = kernel.CreateFileW(str(SCRATCH / 'tmpsvotty7_'), 0x80000000, 7, None, 3, 0x80, None)
    assert log_handle != ctypes.c_void_p(-1).value
    descriptor = msvcrt.open_osfhandle(log_handle, os.O_RDONLY | os.O_BINARY)
    with os.fdopen(descriptor, 'rb') as stream:
        payload = stream.read(collector.MAX_LOG_BYTES + 1)
    assert len(payload) <= collector.MAX_LOG_BYTES
    safe, redactions = collector.redact_credentials(payload.decode('utf-8'))
    with (OUT / 'wheel-log-sanitized.txt').open('x', encoding='utf-8', newline='') as stream:
        stream.write(safe)
    result['log_redactions'] = redactions
    result['artifacts']['wheel-log-sanitized.txt'] = hashlib.sha256((OUT / 'wheel-log-sanitized.txt').read_bytes()).hexdigest()
    receipt = RUN / 'logs/wheel_pytest.process.json'
    (OUT / 'original-wheel-process-receipt.json').write_bytes(receipt.read_bytes())
    save('wheel-boundary-result.json', result)
    print(json.dumps(result), flush=True)
finally:
    kernel.CloseHandle(handle)
