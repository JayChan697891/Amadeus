"""Recover only proven fixture labels after a completed frozen collector row."""
import ast
import copy
from datetime import datetime, timezone
import hashlib
import io
import json
from pathlib import Path
import xml.etree.ElementTree as ET


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def derive(source, run, row, captures):
    from tools import release_acceptance as collector
    from tools.acceptance_evidence import _Evidence, _junit, _verify_process, verify_report_evidence
    report = json.loads((run / 'report.json').read_text(encoding='utf-8'))
    assert report['commit_sha'] == 'b6f05f0473591dc568be25591047fdea963ef86c'
    assert report['working_tree_status'] == '' and report['source_changed_during_execution'] is False
    assert all(c['status'] == 'pass' for c in report['checks'])
    evidence = _Evidence(run, report['artifacts'])
    for name in evidence.members:
        evidence.check(name)
    for check in report['checks']:
        if 'log' in check:
            _verify_process(evidence, check, check['log'])
    fixture = row.parent / 'identity-label-fixture.py.txt'
    assert sha(fixture) == '41e73fb79d993b483021651fe3f832670625710ac1e3a40ef1a203d836fd96df'
    old = ast.parse((source / 'tests/unit/test_release_acceptance.py').read_text(encoding='utf-8'))
    new = ast.parse(fixture.read_text(encoding='utf-8'))
    name = 'test_redaction_is_stable_when_evidence_is_processed_again'
    old_test = next(n for n in old.body if isinstance(n, ast.FunctionDef) and n.name == name)
    new_test = next(n for n in new.body if isinstance(n, ast.FunctionDef) and n.name == name)
    assert sum(k.arg == 'ids' for k in new_test.decorator_list[0].keywords) == 1
    labels = ast.literal_eval(next(k.value for k in new_test.decorator_list[0].keywords if k.arg == 'ids'))
    values = ast.literal_eval(old_test.decorator_list[0].args[1])
    new_test.decorator_list[0].keywords = [k for k in new_test.decorator_list[0].keywords if k.arg != 'ids']
    assert ast.dump(old) == ast.dump(new)
    aliases = {name+'['+value+']':name+'['+label+']' for value,label in zip(values,labels,strict=True)}
    assert len(aliases) == len(set(aliases.values())) == 6
    transformed = {}
    for check_id, filename in (('pytest','junit.xml'),('wheel_pytest','wheel-junit.xml'),('sdist_pytest','sdist-junit.xml')):
        capture = captures[filename]
        assert capture['status'] == 'captured'
        raw = row/'private-junit'/filename
        assert sha(raw) == capture['sha256'] and raw.stat().st_size == capture['size']
        replay = io.BytesIO()
        collector.persist_junit(raw,replay)
        assert replay.getvalue() == (run/filename).read_bytes()
        tree = ET.fromstring(replay.getvalue())
        raw_cases = list(ET.parse(raw).getroot().iter('testcase'))
        saved_cases = list(tree.iter('testcase'))
        assert len(raw_cases) == len(saved_cases)
        unique,mapping = set(),[]
        for ordinal,(original,saved) in enumerate(zip(raw_cases,saved_cases,strict=True)):
            identity = (original.get('classname'),original.get('name'))
            assert identity not in unique
            unique.add(identity)
            if identity[1] in aliases:
                mapping.append({'ordinal':ordinal,'classname':saved.get('classname'),'original_saved_name':saved.get('name'),'recovered_name':aliases[identity[1]],'raw_identity_sha256':hashlib.sha256(json.dumps(identity,separators=(',',':')).encode()).hexdigest()})
                saved.set('name',aliases[identity[1]])
        assert len(mapping) == len(set(m['recovered_name'] for m in mapping)) == 6
        data = ET.tostring(tree,encoding='utf-8',xml_declaration=True)
        counts,case_hash = _junit(data)
        assert counts == next(c['junit'] for c in report['checks'] if c['id']==check_id)
        assert counts['tests'] == len(unique) and counts['failures'] == counts['errors'] == 0
        transformed[filename] = (data,{'check_id':check_id,'raw_sha256':sha(raw),'original_sha256':sha(run/filename),'derived_sha256':hashlib.sha256(data).hexdigest(),'counts':counts,'case_set_sha256':case_hash,'mapping':mapping,'scope':'Only six proven parameter labels changed; all other XML nodes preserved','raw_storage':'Restricted local storage only; never upload'})
    assert len({value[1]['case_set_sha256'] for value in transformed.values()}) == 1
    target=row/'identity-derived';target.mkdir(exist_ok=False)
    def write(name,data):
        path=target/name;path.parent.mkdir(parents=True,exist_ok=True)
        with path.open('xb') as stream:stream.write(data)
        return {'path':name,'sha256':sha(path),'size':len(data)}
    derived=copy.deepcopy(report)
    for name in evidence.members:
        write(name, transformed[name][0] if name in transformed else (run/name).read_bytes())
    additions=[write('identity-original/report.json',(run/'report.json').read_bytes()),write('identity-derivation.py.txt',Path(__file__).read_bytes()),write('identity-label-fixture.py.txt',fixture.read_bytes())]
    for name,(_,proof) in transformed.items():
        additions.append(write('identity-original/'+name,(run/name).read_bytes()))
        proof_name='identity-'+name+'.json'
        additions.append(write(proof_name,(json.dumps(proof,indent=2)+'\n').encode()))
        artifact=next(entry for entry in derived['artifacts'] if entry['path']==name)
        artifact.update(sha256=sha(target/name),size=(target/name).stat().st_size)
    derived['artifacts'].extend(additions)
    derived['identity_recovery']={'scope':'Only six proven fixture labels per completed suite; not a new execution','original_report':additions[0],'created_at':datetime.now(timezone.utc).isoformat(),'formal_release':'BLOCKED'}
    assert {k:v for k,v in derived.items() if k not in ('artifacts','identity_recovery')} == {k:v for k,v in report.items() if k!='artifacts'}
    write('report.json',(json.dumps(derived,indent=2)+'\n').encode())
    audit=verify_report_evidence(target/'report.json',derived,source_repo=source)
    assert audit['junit_case_sets_match'] is True
    write('identity-audit.json',(json.dumps(audit,indent=2)+'\n').encode())
    return target/'report.json',audit
