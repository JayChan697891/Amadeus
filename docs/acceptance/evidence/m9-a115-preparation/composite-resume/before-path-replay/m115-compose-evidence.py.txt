def read(path):
    return json.loads(path.read_text(encoding='utf-8'))


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def pin(path, root=TARGET):
    return {'path': path.relative_to(root).as_posix(), 'sha256': sha(path), 'size': path.stat().st_size}


def save(path, value):
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open('x', encoding='utf-8', newline='\n') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')


def copy_new(source, relative):
    target = TARGET / relative
    target.parent.mkdir(parents=True, exist_ok=True)
    with target.open('xb') as stream:
        stream.write(source.read_bytes())
    return target

# Re-prove the narrow label transformation against the complete original bytes.
old_tree = ast.parse((SOURCE / 'tests/unit/test_release_acceptance.py').read_text(encoding='utf-8'))
fixture = PROOFS / 'tests/unit/test_release_acceptance.py'
assert sha(fixture) == '41e73fb79d993b483021651fe3f832670625710ac1e3a40ef1a203d836fd96df'
new_tree = ast.parse(fixture.read_text(encoding='utf-8'))
old = next(n for n in old_tree.body if isinstance(n, ast.FunctionDef) and n.name == NAME)
new = next(n for n in new_tree.body if isinstance(n, ast.FunctionDef) and n.name == NAME)
assert sum(k.arg == 'ids' for k in new.decorator_list[0].keywords) == 1
labels = ast.literal_eval(next(k.value for k in new.decorator_list[0].keywords if k.arg == 'ids'))
values = ast.literal_eval(old.decorator_list[0].args[1])
new.decorator_list[0].keywords = [k for k in new.decorator_list[0].keywords if k.arg != 'ids']
assert ast.dump(old_tree) == ast.dump(new_tree)
aliases = {NAME + '[' + value + ']': NAME + '[' + label + ']' for value, label in zip(values, labels, strict=True)}
assert len(aliases) == len(set(aliases.values())) == 6
identities = {}
for suite, root, filename, check_id in (
    ('source', ORIGINAL, 'junit.xml', 'pytest'),
    ('sdist', RESUMED / 'run', 'sdist-junit.xml', 'sdist_pytest'),
):
    proof = read(PROOFS / suite / 'proof.json')
    raw = PROOFS / 'private' / (suite + '-junit.xml')
    saved = root / filename
    recovered = PROOFS / suite / 'recovered-junit.xml'
    receipt_path = root / 'logs' / (check_id + '.process.json')
    receipt = read(receipt_path)
    assert receipt['phase'] == 'finished' and receipt['result']['exit_code'] == 0
    assert proof['source_commit'] == COMMIT and proof['suite'] == suite
    assert sha(raw) == proof['raw_sha256'] and sha(saved) == proof['original_persisted_sha256']
    assert sha(recovered) == proof['recovered_sha256'] and sha(receipt_path) == proof['receipt_sha256']
    assert proof['converter_sha256'] == attempt['collector_sha256'] and proof['label_fixture_sha256'] == sha(fixture)
    assert sha(root / receipt['result']['log']) == receipt['result']['log_sha256']
    replay = io.BytesIO()
    collector.persist_junit(raw, replay)
    assert replay.getvalue() == saved.read_bytes()
    expected = ET.fromstring(saved.read_bytes())
    raw_cases = list(ET.parse(raw).getroot().iter('testcase'))
    saved_cases = list(expected.iter('testcase'))
    assert len(raw_cases) == len(saved_cases)
    unique, mapping = set(), []
    for ordinal, (a, b) in enumerate(zip(raw_cases, saved_cases, strict=True)):
        key = (a.get('classname'), a.get('name'))
        assert key not in unique
        unique.add(key)
        if key[1] in aliases:
            mapping.append((ordinal, b.get('classname'), b.get('name'), aliases[key[1]], hashlib.sha256(json.dumps(key, separators=(',', ':')).encode()).hexdigest()))
            b.set('name', aliases[key[1]])
    assert len(mapping) == 6
    assert mapping == [(m['ordinal'], m['classname'], m['original_saved_name'], m['recovered_name'], m['raw_identity_sha256']) for m in proof['mapping']]
    assert ET.tostring(expected) == ET.tostring(ET.parse(recovered).getroot())
    counts, case_hash = _junit(recovered.read_bytes())
    assert counts['tests'] == len(unique) == proof['raw_unique_cases'] == proof['recovered_unique_cases']
    assert counts['failures'] == counts['errors'] == 0
    identities[suite] = (counts, case_hash)
assert identities['source'] == identities['sdist']
